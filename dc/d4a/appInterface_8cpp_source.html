70 AppInterface::EnqueueSegments(const uint8_t* buf, ssize_t n, uint32_t srcIp)
71 {
72  size_t total = static_cast<size_t>(n);
73 
74  // Segment this datagram/packet into fixed-size chunks, padding the last one to m_maxPayloadBytes.
75  // All chunks carry (datagramId, segIndex, totalSegs, datagramTotalBytes, srcIp) for NCC
76  // reassembly and per-UE LLC-queue bucketing.
77  uint16_t totalSegs = static_cast<uint16_t>(
78  (total + m_maxPayloadBytes - 1) / m_maxPayloadBytes);
79  if (totalSegs == 0) totalSegs = 1;
80 
81  {
82  std::lock_guard<std::mutex> lock(m_segmentMutex);
83  for (uint16_t i = 0; i < totalSegs; ++i)
84  {
85  size_t off = static_cast<size_t>(i) * m_maxPayloadBytes;
86  size_t len = std::min(static_cast<size_t>(m_maxPayloadBytes), total - off);
87 
88  std::string segData(m_maxPayloadBytes, '\0');
89  std::memcpy(segData.data(), buf + off, len);
90 
91  m_segmentQueue.push({m_datagramId, i, totalSegs,
92  static_cast<uint32_t>(total), std::move(segData), srcIp});
93  }
94  ++m_datagramId;
95  }
96  m_segmentCv.notify_all();
97 }

132 void
133 AppInterface::TunReceiveLoop()
134 {
135  constexpr size_t BUF_SIZE = 65536;
136  std::vector<uint8_t> buf(BUF_SIZE);
137 
138  while (m_running)
139  {
140  // read() on a TUN fd returns one complete IP packet per call.
141  ssize_t n = read(m_tunFd, buf.data(), BUF_SIZE);
142  if (n <= 0)
143  break;
144 
145  if (m_bitsCollector2)
146  m_bitsCollector2->Collect(static_cast<uint32_t>(n));
147 
148  // Parse IPv4 source address from the IP header — used both for per-UE
149  // statistics below and as the per-UE bucketing key handed to EnqueueSegments
150  // (so the LLC drain stage can fairly interleave distinct UEs' traffic).
151  uint32_t srcIp = 0;
152  if (n >= 20 && (buf[0] >> 4) == 4)
153  {
154  std::memcpy(&srcIp, buf.data() + 12, 4);
155  srcIp = ntohl(srcIp);
156 
157  BitsCollector* ueCollector = nullptr;
158  {
159  std::lock_guard<std::mutex> lock(m_perUeMutex);
160  auto it = m_perUeCollectors.find(srcIp);
161  if (it == m_perUeCollectors.end())
162  {
163  uint8_t lastOctet = static_cast<uint8_t>(srcIp & 0xFF);
164  std::string logName = "ut_app_rx_ue" + std::to_string(lastOctet);
165  BitsCollector* c = MakeBitsCollector(m_logFpath, logName, m_statWindowSize);
166  m_perUeCollectors[srcIp] = c;
167  ueCollector = c;
168  }
169  else
170  {
171  ueCollector = it->second;
172  }
173  }
174  if (ueCollector)
175  ueCollector->Collect(static_cast<uint32_t>(n));
176  }
177 
178  // Feed the raw IP packet bytes through the same TBTP segmentation pipeline.
179  EnqueueSegments(buf.data(), n, srcIp);
180  }
181 }

183 void
184 AppInterface::ProcessLoop()
185 {
186  while (true)
187  {
188  SegmentInfo seg;
189  {
190  std::unique_lock<std::mutex> lock(m_segmentMutex);
191  m_segmentCv.wait(lock, [this]{ return !m_segmentQueue.empty() || !m_running; });
192 
193  if (m_segmentQueue.empty()) // only reachable when !m_running
194  break;
195 
196  seg = std::move(m_segmentQueue.front());
197  m_segmentQueue.pop();
198  }
199 
200  Packet* pkt = new Packet();
201 
202  // Attach segment metadata so NCC can reassemble per-datagram before forwarding to iperf.
203  auto segTag = std::make_unique<SegmentTag>(seg.datagramId, seg.segIndex,
204  seg.totalSegs, seg.datagramTotalBytes);
205  pkt->AddTag(std::move(segTag));
206 
207  TrafficMsg tcMsg;
208  tcMsg.WriteMsg(seg.data);
209  pkt->AddTcMsg(tcMsg);
210  auto tcTag = std::make_unique<TrcTag>();
211  tcTag->SetMsgType(tcMsg.GetTcMsgType());
212  pkt->AddTag(std::move(tcTag));
213 
214  // Tag this packet with its originating UE's source IP rather than the
215  // shared PHY socket: every UE behind this UT funnels through the same
216  // m_phy->GetSock(), so keying the LLC Queue by that value collapses all
217  // UEs into a single bucket and defeats DrainQueueToRawQueue's per-socket
218  // round-robin fairness. Using srcIp as the bucketing key instead makes
219  // each UE a distinct, stable bucket, so the existing round-robin logic
220  // actually interleaves between UEs as intended.
221  int ueQueueKey = static_cast<int>(seg.srcIp);
222  m_llc->Receive(ueQueueKey, pkt);
223 
224  if (m_bitsCollector)
225  m_bitsCollector->Collect(pkt);
226  }
227 }
