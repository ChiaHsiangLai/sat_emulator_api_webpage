71 AppInterface::EnqueueSegments(const uint8_t* buf, ssize_t n, uint32_t srcIp)
72 {
73  size_t total = static_cast<size_t>(n);
74 
75  // Segment this datagram/packet into fixed-size chunks, padding the last one to m_maxPayloadBytes.
76  // All chunks carry (datagramId, segIndex, totalSegs, datagramTotalBytes, srcIp) for NCC
77  // reassembly and per-UE LLC-queue bucketing.
78  uint16_t totalSegs = static_cast<uint16_t>(
79  (total + m_maxPayloadBytes - 1) / m_maxPayloadBytes);
80  if (totalSegs == 0) totalSegs = 1;
81 
82  {
83  std::lock_guard<std::mutex> lock(m_segmentMutex);
84  for (uint16_t i = 0; i < totalSegs; ++i)
85  {
86  size_t off = static_cast<size_t>(i) * m_maxPayloadBytes;
87  size_t len = std::min(static_cast<size_t>(m_maxPayloadBytes), total - off);
88 
89  std::string segData(m_maxPayloadBytes, '\0');
90  std::memcpy(segData.data(), buf + off, len);
91 
92  m_segmentQueue.push({m_datagramId, i, totalSegs,
93  static_cast<uint32_t>(total), std::move(segData), srcIp});
94  }
95  ++m_datagramId;
96  }
97  m_segmentCv.notify_all();
98 }

133 void
134 AppInterface::TunReceiveLoop()
135 {
136  constexpr size_t BUF_SIZE = 65536;
137  std::vector<uint8_t> buf(BUF_SIZE);
138 
139  while (m_running)
140  {
141  // read() on a TUN fd returns one complete IP packet per call.
142  ssize_t n = read(m_tunFd, buf.data(), BUF_SIZE);
143  if (n <= 0)
144  break;
145 
146  if (m_bitsCollector2)
147  m_bitsCollector2->Collect(static_cast<uint32_t>(n));
148 
149  // Parse IPv4 source address from the IP header — used both for per-UE
150  // statistics below and as the per-UE bucketing key handed to EnqueueSegments
151  // (so the LLC drain stage can fairly interleave distinct UEs' traffic).
152  uint32_t srcIp = 0;
153  if (n >= 20 && (buf[0] >> 4) == 4)
154  {
155  std::memcpy(&srcIp, buf.data() + 12, 4);
156  srcIp = ntohl(srcIp);
157 
158  BitsCollector* ueCollector = nullptr;
159  {
160  std::lock_guard<std::mutex> lock(m_perUeMutex);
161  auto it = m_perUeCollectors.find(srcIp);
162  if (it == m_perUeCollectors.end())
163  {
164  uint8_t lastOctet = static_cast<uint8_t>(srcIp & 0xFF);
165  std::string logName = "ut_app_rx_ue" + std::to_string(lastOctet);
166  BitsCollector* c = MakeBitsCollector(m_logFpath, logName, m_statWindowSize);
167  m_perUeCollectors[srcIp] = c;
168  ueCollector = c;
169  }
170  else
171  {
172  ueCollector = it->second;
173  }
174  }
175  if (ueCollector)
176  ueCollector->Collect(static_cast<uint32_t>(n));
177  }
178 
179  // Feed the raw IP packet bytes through the same TBTP segmentation pipeline.
180  EnqueueSegments(buf.data(), n, srcIp);
181  }

184 void
185 AppInterface::ProcessLoop()
186 {
187  while (true)
188  {
189  SegmentInfo seg;
190  {
191  std::unique_lock<std::mutex> lock(m_segmentMutex);
192  m_segmentCv.wait(lock, [this]{ return !m_segmentQueue.empty() || !m_running; });
193 
194  if (m_segmentQueue.empty()) // only reachable when !m_running
195  break;
196 
197  seg = std::move(m_segmentQueue.front());
198  m_segmentQueue.pop();
199  }
200 
201  Packet* pkt = new Packet();
202 
203  // Attach segment metadata so NCC can reassemble per-datagram before forwarding to iperf.
204  auto segTag = std::make_unique<SegmentTag>(seg.datagramId, seg.segIndex,
205  seg.totalSegs, seg.datagramTotalBytes);
206  pkt->AddTag(std::move(segTag));
207 
208  TrafficMsg tcMsg;
209  tcMsg.WriteMsg(seg.data);
210  pkt->AddTcMsg(tcMsg);
211  auto tcTag = std::make_unique<TrcTag>();
212  tcTag->SetMsgType(tcMsg.GetTcMsgType());
213  pkt->AddTag(std::move(tcTag));
214 
215  // Tag this packet with its originating UE's source IP rather than the
216  // shared PHY socket: every UE behind this UT funnels through the same
217  // m_phy->GetSock(), so keying the LLC Queue by that value collapses all
218  // UEs into a single bucket and defeats DrainQueueToRawQueue's per-socket
219  // round-robin fairness. Using srcIp as the bucketing key instead makes
220  // each UE a distinct, stable bucket, so the existing round-robin logic
221  // actually interleaves between UEs as intended.
222  int ueQueueKey = static_cast<int>(seg.srcIp);
223  m_llc->Receive(ueQueueKey, pkt);
224 
225  if (m_bitsCollector)
226  m_bitsCollector->Collect(pkt);
227  }
228 }
