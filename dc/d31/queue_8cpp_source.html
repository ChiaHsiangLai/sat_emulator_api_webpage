295 // Plain round robin: m_scheduler owns turn order (Next()/Report()); this
296 // function drives the round loop and supplies the per-turn policy (does
297 // ueId's head buffer fit the remaining budget?), performing the actual
298 // extraction when it does. Keeps running rounds as long as some UE was
299 // served and the byte budget remains, so one large TX opportunity can serve
300 // many packets, interleaved one-per-UE, across many rounds.
301 void
302 RawQueue::ExtractBufsRoundRobin(uint32_t& txOpportunitySize, std::vector<std::tuple<double, int, std::vector<uint8_t>>>& bufs, int maxBufNum)
303 {
304  uint32_t initialSize = txOpportunitySize;
305  uint32_t remaining = txOpportunitySize;
306  int extracted = 0;
307 
308  auto budgetLeft = [&]() { return maxBufNum < 0 || extracted < maxBufNum; };
309 
310  auto serveUe = [&](int ueId) -> RoundRobinScheduler::Result
311  {
312  auto ueIt = m_ueBuckets.find(ueId);
313  if (ueIt == m_ueBuckets.end() || ueIt->second.bufs.empty())
314  return RoundRobinScheduler::Result::Drained; // bucket already drained
315 
316  if (!budgetLeft())
317  return RoundRobinScheduler::Result::Skipped;
318 
319  UeBucket& bucket = ueIt->second;
320  uint32_t sz = static_cast<uint32_t>(std::get<2>(bucket.bufs.front()).size());
321  if (sz > remaining)
322  return RoundRobinScheduler::Result::Skipped; // doesn't fit this opportunity
323 
324  remaining -= sz;
325  bufs.push_back(std::move(bucket.bufs.front()));
326  bucket.bufs.pop_front();
327  extracted++;
328 
329  if (bucket.bufs.empty())
330  {
331  m_ueBuckets.erase(ueIt);
332  return RoundRobinScheduler::Result::Drained;
333  }
334  return RoundRobinScheduler::Result::Served;
335  };
336 
337  bool progress = true;
338  while (remaining > 0 && !m_scheduler.Empty() && progress && budgetLeft())
339  {
340  progress = false;
341  size_t roundSize = m_scheduler.Size();
342  for (size_t i = 0; i < roundSize; i++)
343  {
344  int ueId = m_scheduler.Next();
345  RoundRobinScheduler::Result result = serveUe(ueId);
346  m_scheduler.Report(ueId, result);
347  if (result == RoundRobinScheduler::Result::Served)
348  progress = true;
349  }
350  }
351 
352  txOpportunitySize = remaining;
353 
354  if (extracted > 0)
355  {
356  std::ostringstream oss;
357  oss << "[" << GetFlowTypeName(m_flow) << " raw queue] Extracted " << extracted
358  << " bufs (round robin), bytes used: " << (initialSize - remaining) << "\n";
359  m_log->WriteToFile(oss.str());
360  }
361 }
