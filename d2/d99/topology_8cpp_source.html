133 Topology::GetElevAngle(double g_lat,double g_lon,double g_alt,double sat_lat,double sat_lon,double sat_alt)
134 {
135  double g_x, g_y, g_z;
136  latlonh_to_ecef(g_lat, g_lon, g_alt, g_x, g_y, g_z);
137 
138  double sat_x, sat_y, sat_z;
139  latlonh_to_ecef(sat_lat, sat_lon, sat_alt, sat_x, sat_y, sat_z);
140 
141  // r = s - u
142  double rx = sat_x - g_x;
143  double ry = sat_y - g_y;
144  double rz = sat_z - g_z;
145 
146  // v = unit vector pointing up (local zenith direction)
147  double u_norm = std::sqrt(g_x*g_x + g_y*g_y + g_z*g_z);
148  double vx = g_x / u_norm;
149  double vy = g_y / u_norm;
150  double vz = g_z / u_norm;
151 
152  // r dot v
153  double r_dot_v = rx*vx + ry*vy + rz*vz;
154 
155  // |r|
156  double r_norm = std::sqrt(rx*rx + ry*ry + rz*rz);
157 
158  // avoid division by zero
159  if (r_norm == 0.0)
160  return 90.0;
161 
162  // Correct relation:
163  // sin(elev) = (r ⋅ v) / |r|
164  double sin_elev = r_dot_v / r_norm;
165 
166  // clamp for safety
167  if (sin_elev > 1.0) sin_elev = 1.0;
168  if (sin_elev < -1.0) sin_elev = -1.0;
169 
170  double elev_rad = std::asin(sin_elev);
171  double elev_deg = elev_rad * 180.0 / M_PI;
172 
173  return elev_deg;
174 }
