109 // Wire size is TagType(4) + tagSize(4) + 4-byte payload = 12 bytes (down
110 // from 16 — the payload itself shrank from a full double to a uint32_t, same
111 // framing convention as every other tag). The payload is milliseconds since
112 // the start of the current SEND_TIME_WRAP_PERIOD_S window rather than an
113 // absolute Unix timestamp, since a raw 4-byte float can't hold both
114 // Unix-epoch-scale magnitude and sub-second precision at once.
115 static constexpr double SEND_TIME_WRAP_PERIOD_S = 86400.0; // 1 day; must exceed the largest one-way delay ever measured

118 SendTimeTag::Serialize(std::vector<uint8_t>& buffer) const
119 {
120  SerializerUtils::Write(buffer, static_cast<uint32_t>(GetTagType()));
121  double posInPeriod = std::fmod(m_sendTime, SEND_TIME_WRAP_PERIOD_S);
122  if (posInPeriod < 0) posInPeriod += SEND_TIME_WRAP_PERIOD_S;
123  uint32_t msInPeriod = static_cast<uint32_t>(posInPeriod * 1000.0 + 0.5);
124  SerializerUtils::Write(buffer, static_cast<uint32_t>(sizeof(msInPeriod)));

129 SendTimeTag::Deserialize(const std::vector<uint8_t>& buffer, size_t& offset)
130 {
131  offset += sizeof(uint32_t); // Skip TagType
132 
133  uint32_t tagSize;
134  SerializerUtils::Read(buffer, offset, tagSize);
135 
136  if (offset + tagSize > buffer.size()) {
137  throw std::runtime_error("SendTimeTag: buffer overflow");
138  }
139 
140  uint32_t msInPeriod;
141  SerializerUtils::Read(buffer, offset, msInPeriod);
142 
143  // Reconstruct an absolute timestamp by picking whichever period window
144  // (this one, or the previous one) puts the send time at or before now —
145  // correct as long as the packet's actual one-way delay is less than
146  // SEND_TIME_WRAP_PERIOD_S, which the receiver's own clock is used for.
147  double now = Time::Now();
148  double periodStart = std::floor(now / SEND_TIME_WRAP_PERIOD_S) * SEND_TIME_WRAP_PERIOD_S;
149  double candidate = periodStart + msInPeriod / 1000.0;
150  if (candidate > now + 1.0)
151  candidate -= SEND_TIME_WRAP_PERIOD_S;
152  m_sendTime = candidate;
153 }
