585  // serialization; packets were serialized by PreSerializeWorker as they
586  // arrived in the BE LLC Queue).
587  FlowType flowType = (rc == 0) ? FlowType::CONTROL_FLOW : FlowType::BE_FLOW;
588  RawQueue* rawQueue = m_llc->GetRawQueue(flowType);
589 
590  bool isTcSlot = (rc == 0) ? false : true;

593  std::vector<std::tuple<double, int, std::vector<uint8_t>>> bufs;
594  rawQueue->ExtractBufs(remaining, bufs);
595 
596  if (!bufs.empty())
597  {
598  std::vector<uint8_t> txBuf;
599  txBuf.reserve(bytes - remaining);
600  double now = Time::Now();
601  std::map<int, std::ofstream> delayLogs;
602  for (auto& [enqueueTime, ueId, b] : bufs)
603  {
604  txBuf.insert(txBuf.end(), b.begin(), b.end());

1510  SendHandoverNotificationMsg(m_phy->GetSock());
