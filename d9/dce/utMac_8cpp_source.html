1766 // ─────────────────────────────────────────────────────────────────────────────
1767 // Pre-serialization: drain LLC Queue → serialize → push to RawQueue.
1768 //
1769 // This worker no longer decides *which UE* goes first — it just demuxes and
1770 // serializes everything the LLC Queue has, tagged with its originating UE id.
1771 // The fairness decision (who gets served, and how much) is made later, at TX
1772 // time, by RawQueue's Deficit Round Robin extraction (see queue.cpp). That
1773 // split keeps the expensive part (serialization) off the real-time TX path
1774 // while keeping the fairness decision as close to the actual TX opportunity
1775 // as possible.
1776 // ─────────────────────────────────────────────────────────────────────────────
1777 
1778 // Extract every available packet from the given LLC Queue, serialize it to a
1779 // byte buffer tagged with its originating UE id, and push it into the
1780 // companion RawQueue. Stats and tracing happen at actual TX time in
1781 // TransmitBitStream.
1782 void
1783 UtMAC::DrainQueueToRawQueue(FlowType flow)
1784 {
1785  Queue* queue = m_llc->GetQueue(flow);
1786  RawQueue* rawQueue = m_llc->GetRawQueue(flow);
1787 
1788  while (!queue->IsEmpty())
1789  {
1790  std::vector<std::pair<int, Packet*>> pkts;
1791  queue->ExtractPacket(pkts); // extract one packet at a time
1792  for (auto& [ueId, pkt] : pkts)
