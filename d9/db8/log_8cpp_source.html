16  // Shared background threads that drain every async file logger created by
17  // any LOG instance in this process. Must exist before the first
18  // create_async_nb() call.
19  //
20  // Sized generously (large queue, several workers) because this process
21  // can have very many independently-paced loggers -- some firing every
22  // packet (MacLogFile/PhyLogFile/LlcLogFile), others on a steady 1Hz timer
23  // (BitsCollector::SaveData's per-UT/per-flow throughput files). A small
24  // shared queue drained by a single thread let the high-frequency loggers
25  // starve the 1Hz ones of queue space indefinitely: their WriteToFile
26  // calls (create_async's default overflow policy blocks the caller when
27  // the queue is full) never won a free slot again, so those specific
28  // files silently stopped gaining new lines partway through a run while
29  // everything else kept logging fine -- see GetLogger()'s use of
30  // create_async_nb below for the other half of this fix.
31  void EnsureThreadPoolInitialized()
32  {
33  static bool initialized = []() {
34  spdlog::init_thread_pool(1 << 20, 4);
35  return true;
36  }();
37  (void)initialized;
38  }

40  // spdlog's file sink appends its own end-of-line after every log() call.
41  // Callers here already bake a trailing "\n" into oss, so strip it first --
42  // otherwise every line would come out double-spaced, breaking the exact
43  // on-disk format that existing log-parsing tooling depends on.
44  std::string StripTrailingNewline(std::string s)
45  {
46  while (!s.empty() && (s.back() == '\n' || s.back() == '\r'))
47  s.pop_back();
48  return s;
49  }

106 LOG::GetLogger(const std::string& fpath, const std::string& name)
107 {
108  std::string fullPath = fpath + name;
109 
110  std::lock_guard<std::mutex> lock(m_mapMutex);
111  auto it = m_loggerMap.find(fullPath);
112  if (it != m_loggerMap.end())
113  return it->second;
114 
115  // First use of this path: spdlog's file sink will create the file itself,
116  // but not its parent directory.
117  if (!fs::exists(fpath))
118  {
119  fs::create_directories(fpath);
120  }
121 
122  EnsureThreadPoolInitialized();
123 
124  // spdlog keeps a process-wide registry keyed by name; check it first so a
125  // second LOG instance targeting the same path reuses the existing logger
126  // instead of throwing on a duplicate registration.
127  std::shared_ptr<spdlog::logger> logger = spdlog::get(fullPath);
128  if (!logger)
129  {
130  // create_async_nb (overrun_oldest), not create_async (block): if the
131  // shared queue is ever momentarily full, drop the oldest buffered
132  // line instead of blocking this call's caller. Blocking here is what
133  // caused the real stall this fix addresses -- see
134  // EnsureThreadPoolInitialized() above for the full explanation.
135  // Losing an occasional line under a genuine burst is an acceptable
136  // trade against a logger silently never writing again.
137  logger = spdlog::create_async_nb<spdlog::sinks::basic_file_sink_mt>(fullPath, fullPath, /*truncate=*/false);
138  // Raw message only: callers pre-format their own "[Y-m-d H:M:S.mmm] ..."
139  // timestamp, so this keeps the on-disk format identical to before
140  // spdlog was introduced.
141  logger->set_pattern("%v");
142  }
143 
144  // spdlog's basic_file_sink uses a buffered std::ofstream. A low-rate
145  // statistic file may never fill that buffer before the runner ends by
146  // SIGKILL, leaving it visibly truncated at exactly 4096 bytes. Request
147  // a flush for those files while preserving batched async I/O for the
148  // high-frequency protocol logs.
149  if (m_flushOnWrite)
