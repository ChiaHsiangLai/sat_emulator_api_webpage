10 def _read_frame_carrier_counts(net_config_path: str):
11  """
12  Parse Net_Configurations.txt and return (num_frames, carrier_counts),
13  computed the same way FrameConf::GetCarrierCount() does in the C++
14  simulator (model/frameConf.cpp): carrierCount = FrameXBandwidth /
15  FrameXCarrierBandwidth, for X in 0..FrameCount-1. This is the TRUE
16  physical frequency plan -- every carrier that actually exists for that
17  frame, regardless of whether this particular superframe happened to
18  allocate a slot on all of them.
19 
20  Returns (None, None) if the file doesn't exist or is missing the keys
21  needed, so callers can fall back to auto-detecting from the data.
22  """
23  if not net_config_path or not os.path.exists(net_config_path):
24  return None, None
25 
26  cfg = {}
27  with open(net_config_path, "r") as f:
28  for line in f:
29  line = line.strip()
30  if not line or line.startswith("#") or "=" not in line:
31  continue
32  key, value = line.split("=", 1)
33  cfg[key.strip()] = value.strip()
34 
35  if "FrameCount" not in cfg:
36  return None, None
37 
38  try:
39  frame_count = int(cfg["FrameCount"])
40  carrier_counts = []
41  for f in range(frame_count):
42  bw = float(cfg[f"Frame{f}Bandwidth"])
43  carrier_bw = float(cfg[f"Frame{f}CarrierBandwidth"])
44  carrier_counts.append(max(1, round(bw / carrier_bw)))
45  return frame_count, carrier_counts
46  except (KeyError, ValueError, ZeroDivisionError):
47  return None, None
