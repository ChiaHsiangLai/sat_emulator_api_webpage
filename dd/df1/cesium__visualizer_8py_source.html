12 def _real_user_xauthority():
13  """Locate the actual logged-in desktop user's X11 auth cookie file.
14 
15  This whole simulation runs under sudo, which resets $HOME to /root by
16  default. X11 clients look up their connection auth cookie at
17  $HOME/.Xauthority unless XAUTHORITY is set explicitly, so a Chrome
18  launched by this (root) process with no override tries root's own
19  (nonexistent, or plain wrong) Xauthority -- producing "Authorization
20  required, but no authorization protocol specified" and a GPU-process
21  init failure, even though DISPLAY itself is fine. SUDO_USER is the
22  same mechanism _default_base_dir() already relies on to recover the
23  real invoking user's identity when running as root.
24 
25  Checks the GDM/systemd runtime-dir location before the traditional
26  $HOME/.Xauthority: verified empirically on this exact machine (Ubuntu +
27  GNOME/GDM, per OBS's own "Desktop Environment: ubuntu:GNOME" log line)
28  that the active cookie actually lives at /run/user/<uid>/gdm/Xauthority
29  -- $HOME/.Xauthority either doesn't exist there at all, or (worse,
30  found the hard way) can belong to a completely different, unrelated
31  user left over on a shared machine, which would silently hand Chrome
32  the wrong cookie instead of just failing loudly.
33  """
34  sudo_user = os.environ.get("SUDO_USER")
35  if sudo_user:
36  try:
37  pw = pwd.getpwnam(sudo_user)
38  home, uid = pw.pw_dir, pw.pw_uid
39  except KeyError:
40  home, uid = os.path.expanduser(f"~{sudo_user}"), None
41  else:
42  home, uid = os.path.expanduser("~"), os.getuid()
43 
44  candidates = []
45  if uid is not None:
46  candidates.append(f"/run/user/{uid}/gdm/Xauthority")
47  candidates.append(os.path.join(home, ".Xauthority"))
48 
49  for candidate in candidates:
50  if os.path.exists(candidate):
51  return candidate
52  return None

55 def _grant_root_x_access():
56  """Ask the X server to accept connections from local root, via the
57  real desktop user's already-working session -- instead of trying to
58  hand root a copy of the right auth cookie.
59 
60  _real_user_xauthority()'s approach (guess where the cookie file lives,
61  point XAUTHORITY at it) turned out not to be reliable enough on its
62  own: display managers differ (GDM's /run/user/<uid>/gdm/Xauthority is
63  only one convention among several), so a guessed path can miss
64  entirely and this (root, via sudo) process is left with no valid
65  cookie -- reproducing "Authorization required, but no authorization
66  protocol specified" even after the guess-the-path fix.
67 
68  xhost sidesteps the whole problem: it's a *host-based* ACL on the X
69  server ("also accept local root"), not a cookie-matching problem, so
70  it doesn't matter which Xauthority convention this desktop uses. The
71  trick is *who* runs it -- `sudo -u <real user> xhost ...` executes as
72  that user, who already has a working session (correct HOME/cookie by
73  construction, since their desktop set it up), so this one call doesn't
74  hit the same chicken-and-egg problem root calling X directly does.
75  Best-effort: xhost may be missing, or this may not even be a plain
76  Xorg/xhost-compatible session -- failure here just means the
77  XAUTHORITY fallback in _real_user_xauthority() is all that's left,
78  not a hard error for the whole visualizer.
79  """
80  sudo_user = os.environ.get("SUDO_USER")
81  if not sudo_user:
82  return
83  try:
84  subprocess.run(
85  ["sudo", "-u", sudo_user, "xhost", "+SI:localuser:root"],
86  capture_output=True, text=True, timeout=5,
87  )
88  except Exception:
89  pass

92 class CesiumVisualizer:
93  """Renders the same live satellite/UT feed as SatelliteVisualizer
94  (satellite_visualizer.py) in a real browser via CesiumJS instead of a
95  local matplotlib window. Consumes the identical data_queue schema (see
96  dynamic_link_control's dat_queue.put(...) in simple_sns3_test_v14.py):
97  each item is
98  {'sat_id', 'lat', 'lon', 'alt', 'uts_info': [{'id','lat','lon','alt','elev','status'}, ...]}
99 
100  Unlike SatelliteVisualizer -- which is designed to be handed to a
101  separate multiprocessing.Process because matplotlib's
102  FuncAnimation/plt.show() blocks the whole process -- this runs entirely
103  in background threads of the calling process: a browser window is
104  already its own separate OS process, so there's nothing here that needs
105  isolating the same way.
106 
107  Bridging mechanism: a background thread drains data_queue and writes
108  the accumulated state to a small JSON file; a plain local HTTP server
109  (same idea as the http.server usage already used elsewhere in this
110  project for HLS/webpage serving) serves that JSON file plus the
111  companion cesium_visualizer.html, which polls the JSON on a timer and
112  updates the Cesium entities. No websockets/extra pip dependency needed.
113  """

186  # Runs on the host directly (not through a mininet node's
187  # net.get(...).cmd()), same as SatelliteVisualizer's matplotlib
188  # window -- this is a visualization tool for the operator, not
189  # simulated network traffic, so it doesn't need to go through
190  # any mininet namespace. --no-sandbox matches the flag already
191  # used for the webpage-viewer google-chrome launches elsewhere
192  # in this project (this whole script runs under sudo).
193  #
194  # Deliberately NOT reusing the webpage-viewer's --disable-gpu
195  # --disable-software-rasterizer flags: those exist to avoid GPU
196  # compositing flicker on a plain 2D page shared across a virtual
197  # X display, but CesiumJS needs an actual working WebGL context.
198  # --disable-gpu removes the GPU-accelerated path and
199  # --disable-software-rasterizer removes Chrome's SwiftShader
200  # software fallback -- combined, they leave *no* path left to
201  # create a WebGL context at all, which is exactly the "browser
202  # supports WebGL, but initialization failed" error this
203  # produced.
204  #
205  # --use-angle=swiftshader-webgl forces Chrome onto its bundled
206  # software GL renderer unconditionally (verified empirically:
207  # confirmed against an actual Chrome install -- the older
208  # --use-gl=swiftshader spelling is rejected by current
209  # Chrome/ANGLE with "Requested GL implementation (gl=none,
210  # angle=none) not found in allowed implementations", which
211  # reproduces this exact CesiumWidget failure; --use-angle=
212  # swiftshader-webgl is the spelling this Chrome version
213  # actually accepts). This doesn't depend on whatever real
214  # GPU/driver setup happens to exist on this (possibly
215  # virtual/shared-display) host.
216  # --enable-unsafe-swiftshader: Chrome logs this exact flag name
217  # as what it wants when falling back to software WebGL ("Please
218  # use the --enable-unsafe-swiftshader flag to opt in..."),
219  # observed even with --use-angle=swiftshader-webgl already set --
220  # harmless here since this page only ever renders trusted local
221  # content, never arbitrary web pages.
222  #
223  # The remaining flags bring back the webpage-viewer's
224  # anti-flicker set (see launch_uts_and_ues's google-chrome
225  # launch for 'webpage' UE_TRAFFIC_TYPE) for the *window
226  # compositing* layer specifically -- distinct from the WebGL
227  # context layer the flags above are about. Whole-window
228  # black/white flicker (confirmed against a real physical
229  # display, not a remote-desktop artifact) is what those flags
230  # were originally added to prevent on this project's shared/
231  # virtual X display, for plain 2D pages. None of them disable
232  # the GPU process itself (that's --disable-gpu, which breaks
233  # WebGL entirely -- see above), only its use for *compositing*
234  # the browser's own window, so this should coexist with the
235  # swiftshader-webgl WebGL path rather than fight it.
236  # D-Bus session-bus errors (dbus/bus.cc "Did not receive a
237  # reply") are left alone deliberately: D-Bus enforces its

240  # so there's no XAUTHORITY-style fix available for a process
241  # that's genuinely root. Purely cosmetic log noise -- verified
