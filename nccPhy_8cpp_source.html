163 NccPhy::Receive(int clientSock, std::string rx_type)
164 {
165  const uint32_t MAX_ALLOWED_SIZE = 10 * 1024 * 1024; // guard against abnormally large packets (10 MB)
166 
167  while (true) {
168  // --- Step 1: read header (4 bytes) ---
169  uint32_t netLen = 0;
170  size_t headerRead = 0;
171  uint8_t* headerPtr = reinterpret_cast<uint8_t*>(&netLen);
172 
173  while (headerRead < sizeof(uint32_t)) {
174  ssize_t n = recv(clientSock, headerPtr + headerRead, sizeof(uint32_t) - headerRead, 0);
175  if (n <= 0) {
176  m_log->WriteToFile("[NCC PHY] Connection closed while reading header.\n");
177  return;
178  }
179  headerRead += n;
180  }
181 
182  // --- Step 2: convert length and validate ---
183  // uint32_t len = ntohl(netLen);
184  uint32_t len = netLen;
185 
186  if (len == 0 || len > MAX_ALLOWED_SIZE) {
187  m_log->WriteToFile("[NCC PHY] Error: Invalid packet length received: " + std::to_string(len) + "\n");
188  // Invalid length usually means stream sync is broken; disconnect or return error
189  return;
190  }
191 
192  // --- Step 3: read payload by length ---
193  std::vector<uint8_t> buf(len);
194  size_t bodyReceived = 0;
195  while (bodyReceived < len) {
196  ssize_t n = recv(clientSock, buf.data() + bodyReceived, len - bodyReceived, 0);
197  if (n <= 0) {
198  m_log->WriteToFile("[NCC PHY] Connection closed while reading payload.\n");
199  return;
200  }
201  bodyReceived += n;
202  }
203 
204  // --- Step 4: successfully received, dispatch data ---
205  m_log->WriteToFile("[NCC PHY] Received packet size: " + std::to_string(len) + " success.\n");
206 
207  double cno = 0.0;
208  if (rx_type == "TcRx")
209  m_nccMac->ReceiveTcPkt(clientSock, buf, cno);
210  else
211  m_nccMac->Receive(clientSock, buf, cno);
212  }
