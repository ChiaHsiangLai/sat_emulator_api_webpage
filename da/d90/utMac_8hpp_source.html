24 // Fixed-size thread pool: N worker threads pick tasks from a shared queue.
25 // ─────────────────────────────────────────────────────────────────────────────
26 class TxThreadPool
27 {
28 public:
29  // Spawn n worker threads that drain m_tasks until Stop.
30  explicit TxThreadPool(size_t n)
31  {
32  for (size_t i = 0; i < n; ++i)
33  m_workers.emplace_back([this] {
34  while (true) {
35  std::function<void()> task;
36  {
37  std::unique_lock<std::mutex> lk(m_mu);
38  m_cv.wait(lk, [this]{ return m_stop || !m_tasks.empty(); });
39  if (m_stop && m_tasks.empty()) return;
40  task = std::move(m_tasks.front());
41  m_tasks.pop();
42  }
43  task();
44  }
45  });
46  }
47 
48  // Signal stop and join all workers.
49  ~TxThreadPool()
50  {
51  { std::lock_guard<std::mutex> lk(m_mu); m_stop = true; }
52  m_cv.notify_all();
53  for (auto& w : m_workers) if (w.joinable()) w.join();
54  }
55 
56  // Disable copy/move
57  TxThreadPool(const TxThreadPool&) = delete;
58  TxThreadPool& operator=(const TxThreadPool&) = delete;
59 
60  // Push a callable task onto the queue; wakes one worker.
61  template<class F>
62  void Enqueue(F&& f)
63  {
64  { std::lock_guard<std::mutex> lk(m_mu); m_tasks.emplace(std::forward<F>(f)); }
65  m_cv.notify_one();
66  }
67 
68 private:
69  std::vector<std::thread> m_workers; // pool threads
70  std::queue<std::function<void()>> m_tasks; // pending callables
71  std::mutex m_mu; // protects m_tasks and m_stop
72  std::condition_variable m_cv; // signals task availability or stop
73  bool m_stop{false}; // true after destructor
74 };

77 // Precision scheduler: one thread sleeps until the earliest deadline, then
78 // dispatches the task to TxThreadPool. Supports early wake-up when a new
79 // item with a sooner deadline is enqueued.
80 // ─────────────────────────────────────────────────────────────────────────────
81 class TxScheduler
82 {
83 public:

85  explicit TxScheduler(size_t poolSize = 4)
86  : m_pool(poolSize)
87  , m_running(true)
88  , m_schedThread(&TxScheduler::SchedLoop, this)
89  {}

91  // Stop scheduling loop and join the scheduler thread.
92  ~TxScheduler()
93  {
94  { std::lock_guard<std::mutex> lk(m_mu); m_running = false; }
95  m_cv.notify_one();
96  if (m_schedThread.joinable()) m_schedThread.join();
97  }

102  // Schedule task to run at (or after) deadline.
103  void Schedule(std::chrono::steady_clock::time_point deadline,
104  std::function<void()> task)
105  {
106  {
107  std::lock_guard<std::mutex> lk(m_mu);
108  m_pq.push({deadline, std::move(task)});
109  }
110  m_cv.notify_one(); // wake scheduler to re-evaluate earliest deadline
111  }
112 
113  // Drain the queue without executing pending tasks (used during handover reset).
114  void Clear()
115  {
116  std::lock_guard<std::mutex> lk(m_mu);
117  while (!m_pq.empty()) m_pq.pop();
118  }
119 
120 private:
121  // Priority queue entry pairing a deadline with its callable.
122  struct Item {
123  std::chrono::steady_clock::time_point deadline;
124  std::function<void()> task;
125  bool operator>(const Item& o) const { return deadline > o.deadline; }
126  };
127 
128  // Main loop: sleep until the next deadline, then dispatch all expired items.
129  void SchedLoop()
130  {
131  while (true)
132  {
133  std::unique_lock<std::mutex> lk(m_mu);
134 
135  if (!m_running) break;
136 
137  if (m_pq.empty()) {
138  m_cv.wait(lk, [this]{ return !m_running || !m_pq.empty(); });
139  continue;
140  }
141 
142  auto nextDeadline = m_pq.top().deadline;
143 
144  // Sleep until deadline OR a new item arrives (notify_one wakes us early).
145  m_cv.wait_until(lk, nextDeadline, [this, &nextDeadline]{
146  return !m_running
147  || (!m_pq.empty() && m_pq.top().deadline < nextDeadline);
148  });
149 
150  if (!m_running) break;
151 
152  auto now = std::chrono::steady_clock::now();
153  while (!m_pq.empty() && m_pq.top().deadline <= now)
154  {
155  auto item = m_pq.top(); m_pq.pop();
156  lk.unlock();
157  m_pool.Enqueue(std::move(item.task));
158  lk.lock();
159  now = std::chrono::steady_clock::now();
160  }
161  }
162  }
163 
164  TxThreadPool m_pool; // worker threads that execute dispatched tasks
165  std::priority_queue<Item, std::vector<Item>, std::greater<Item>> m_pq; // min-heap by deadline
166  std::mutex m_mu; // protects m_pq and m_running
167  std::condition_variable m_cv; // wakes SchedLoop on new item or stop
168  bool m_running; // false after destructor
169  std::thread m_schedThread; // runs SchedLoop
170 };
