1 #!/usr/bin/env python3
2 """
3 web_page_client.py — simulates a browser loading a web page: fetches
4 index.html, discovers its embedded resources (href/src references), then
5 fetches all of them over <concurrency> persistent connections — the same
6 parallel-connection-pool behavior a real browser uses on page load (e.g. 6
7 concurrent connections, each reused for several sequential requests), rather
8 than opening a brand new connection per object.
9 
10 Resources are split round-robin into <concurrency> groups; each group is
11 fetched via a single curl invocation chained with `--next`, which reuses the
12 same TCP connection for every URL in that group instead of reconnecting each
13 time. Groups run concurrently. There is no pacing: a real page load fetches
14 everything as fast as possible, so the total elapsed time is a genuine
15 page-load-time measurement.
16 
17 Each object within a group is individually bounded by --speed-limit /
18 --speed-time (STALL_SPEED_BYTES_PER_SEC / STALL_WINDOW_S): only a sustained
19 near-zero-throughput stall gives up on that object, moving on to the next
20 --next stage (reusing the connection if it's still alive, reconnecting if
21 not). This is deliberately speed-based rather than a fixed total-time cap —
22 a large object legitimately needs longer to transfer even when everything is
23 healthy, and a fixed cap can't tell "big but fine" apart from "actually
24 stuck" without scaling with WEB_PAGE_SIZE_MB. A large-but-genuinely-
25 transferring object is therefore allowed to keep going as long as it needs
26 (up to the whole group's own time budget), while a real stall is still
27 detected and moved past.
28 
29 Results stream out in real time as each object finishes, rather than
30 appearing all at once when a whole group completes. This requires two
31 things: (1) curl is run under `stdbuf -oL` to force line-buffered stdout —
32 by default curl fully buffers stdout when it isn't a TTY (as it is here,
33 piped to Python), so its -w output for completed objects would otherwise sit
34 in curl's own buffer and only appear once the whole multi-object chain exits;
35 (2) fetch_group reads that output incrementally (via select() on the pipe)
36 and reports each line through a callback the moment it arrives, instead of
37 capturing the whole process's output in one blocking subprocess.run() call.
38 
39 NOTE: this project previously tried HTTP/1.1 keep-alive against Python's
40 plain http.server for an earlier HLS video client and hit
41 ConnectionResetError / BrokenPipeError under this same mininet topology.
42 web_page_server.py enables keep-alive the same way — if requests here start
43 failing with connection resets instead of stalls, that's the same known
44 rough edge resurfacing, not a new bug.
45 
46 Loads the page (index.html + every discovered resource) over and over —
47 like a user repeatedly reloading/browsing — until <duration_s> elapses, not
48 just once. On exit (whether the run_duration budget ran out between page
49 loads, mid-page-load, or the page never became available), writes a small
50 key=value summary to <summary_path> with a status/elapsed_s pattern, plus a
51 page_loads_completed count and object totals accumulated across every page
52 load in this run.
53 
54 Usage:
55  web_page_client.py <page_url> <duration_s> <summary_path> [<concurrency>]
56  e.g. web_page_client.py http://10.0.255.254:5099/index.html 300 \\
57  log_file/UE_10.0.200.101/web_page_duration.txt 6
58 """

69 CONNECT_TIMEOUT_S = 60 # measured ut_queueing_delay averages ~15.6s and peaks ~70s under
70 # congestion — a SYN/SYN-ACK genuinely waiting its turn in that same queue can easily need
71 # more than a few seconds; 5s just guaranteed most connection attempts would be killed
72 # before ever getting a chance, no matter how many times they're retried. Retrying doesn't
73 # shrink the queue, so the fix is to wait long enough for a fair shot instead.
74 
75 # Stall detection is speed-based, not total-time-based: a fixed --max-time cap would need
76 # to scale with WEB_PAGE_SIZE_MB (bigger objects legitimately need longer, even at full
77 # health) — there's no single constant that's both "generous enough for a big object" and
78 # "quick to give up on a truly dead one". --speed-limit/--speed-time instead aborts only
79 # when throughput has been below STALL_SPEED_BYTES_PER_SEC for STALL_WINDOW_S straight
80 # seconds — a real stall — regardless of the object's size, letting a large-but-genuinely-
81 # transferring object run as long as it needs (up to the whole group's own time budget).
82 STALL_SPEED_BYTES_PER_SEC = 50
83 STALL_WINDOW_S = 30

105 def fetch_text(url):
106  """GET url via curl and return the response body as text, or "" if curl
107  failed. Without checking the exit code, a 404 (e.g. before the server
108  has the page ready yet) still returns a non-empty HTML error body

110  result = subprocess.run(

113  )
114  if result.returncode != 0:

119 def fetch_group(urls, timeout_s, on_result, stop_event):
120  """Fetch urls sequentially over a single curl invocation, chained with
121  --next so the TCP connection is reused across all of them instead of
122  reconnecting per object. Each object is individually bounded by
123  --speed-limit/--speed-time (a real stall, sustained near-zero throughput,
124  gives up on just that object and moves to the next --next stage) rather
125  than a fixed --max-time, so a large-but-genuinely-transferring object
126  isn't cut off early just for being big.
127 
128  Calls on_result(url, elapsed_s_or_None) the moment each object's result
129  is available (elapsed_s_or_None is None for a failure: non-200 response,
130  curl error, connect-timeout, or a sustained stall) — in the same order as
131  urls. Any urls whose result never arrives because the group ran out of
132  time (timeout_s) or stop_event got set are reported as None too."""
133  cmd = ["stdbuf", "-oL", "curl"]
134  for i, url in enumerate(urls):
135  if i > 0:
136  cmd.append("--next")
137  cmd += ["-s", "--connect-timeout", str(CONNECT_TIMEOUT_S),
138  "--speed-limit", str(STALL_SPEED_BYTES_PER_SEC), "--speed-time", str(STALL_WINDOW_S),
139  "-o", "/dev/null", "-w", "%{http_code} %{time_total}\n", url]
140 
141  proc = subprocess.Popen(cmd, stdout=subprocess.PIPE)
142  deadline = time.time() + timeout_s
143  idx = 0
144  buf = b""
145  try:
146  while idx < len(urls) and not stop_event.is_set():
147  remaining = deadline - time.time()
148  if remaining <= 0:
149  break
150  ready, _, _ = select.select([proc.stdout], [], [], min(remaining, POLL_INTERVAL_S))
151  if not ready:
152  continue # nothing new yet — loop back to re-check stop_event/deadline
153  chunk = os.read(proc.stdout.fileno(), 4096)
154  if not chunk:
155  break # curl exited — EOF
156  buf += chunk
157  while b"\n" in buf and idx < len(urls):
158  line, buf = buf.split(b"\n", 1)
159  parts = line.decode(errors="replace").split()
160  if len(parts) == 2 and parts[0] == "200":
161  on_result(urls[idx], float(parts[1]))
162  else:
163  on_result(urls[idx], None)
164  idx += 1
165  finally:
166  if proc.poll() is None:
167  proc.kill()
168  proc.wait()
169 
170  for i in range(idx, len(urls)):
171  on_result(urls[i], None)

174 def load_page_once(page_url, base_url, concurrency, run_duration, start, page_num):
175  """Load the page (index.html + every discovered resource) exactly once.
176  Returns (total, fetched, failed, cut_short) — cut_short is True if
177  run_duration ran out before this page load finished (including while
178  still waiting for index.html itself)."""
179  html = ""
180  while time.time() - start < run_duration:
181  html = fetch_text(page_url)
182  if html:
183  break
184  print(f"[web_page_client] page #{page_num}: empty/unreachable page at {page_url}, retrying in 2s",
185  flush=True)
186  time.sleep(2)
187 
188  if not html:
189  print(f"[web_page_client] page #{page_num}: page never became available at {page_url}, giving up",
190  flush=True)
191  return 0, 0, 0, True
192 
193  resource_names = RESOURCE_RE.findall(html)
194  resource_urls = [urllib.parse.urljoin(base_url, name) for name in resource_names]
195  total = len(resource_urls)
196  print(f"[web_page_client] page #{page_num}: fetched index page, {total} embedded resources found",
197  flush=True)
198 
199  # Round-robin split into <concurrency> groups, each fetched over its own
200  # reused connection (see fetch_group) — this is the "N persistent
201  # connections" model instead of "N simultaneous but throwaway
202  # connections", cutting per-object TCP handshake overhead.
203  groups = [g for g in (resource_urls[i::concurrency] for i in range(concurrency)) if g]
204 
205  counts_lock = threading.Lock()
206  fetched = 0
207  failed = 0
208 
209  def on_result(url, fetch_time):
210  nonlocal fetched, failed
211  name = os.path.basename(url)
212  with counts_lock:
213  if fetch_time is None:
214  failed += 1
215  print(f"[web_page_client] page #{page_num}: failed to fetch {name} "
216  f"(curl error, connect-timeout, or stall)", flush=True)
217  else:
218  fetched += 1
219  print(f"[web_page_client] page #{page_num}: fetched {name} (took {fetch_time:.2f}s)",
220  flush=True)
221 
222  # Bound each group by the test's own time budget rather than an
223  # arbitrary fixed cap — under congestion, a group can genuinely take a
224  # long time to work through (queueing delay observed well into the tens
225  # of seconds). stop_event lets this function cut every group short the
226  # moment run_duration is exceeded, without waiting for each one's own
227  # deadline.
228  group_timeout = CONNECT_TIMEOUT_S + run_duration
229  stop_event = threading.Event()
230  pool = concurrent.futures.ThreadPoolExecutor(max_workers=len(groups))
231  futures = [pool.submit(fetch_group, g, group_timeout, on_result, stop_event) for g in groups]
232 
233  remaining = max(run_duration - (time.time() - start), 0)
234  _, not_done = concurrent.futures.wait(futures, timeout=remaining)
235 
236  cut_short = bool(not_done)
237  if cut_short:
238  stop_event.set()
239  concurrent.futures.wait(not_done) # give the still-running groups a moment to notice stop_event and exit
240 
241  pool.shutdown(wait=True)
242  return total, fetched, failed, cut_short
