160 def beam_update(lon_c, lat_c, R_earth_km, beam_radius_deg=beam_radius_deg, num_points=100):
161  """計算波束在地表覆蓋的圓形 ECEF 坐標"""
162  # 將角度轉換為弧度
163  r_beam = np.radians(beam_radius_deg)
164  phi_c = np.radians(lat_c)
165  lambda_c = np.radians(lon_c)
166  theta = np.linspace(0, 2*np.pi, num_points)
167 
168  # Great Circle Distance 公式應用於計算波束圓周
169  phi_circle = np.arcsin(np.sin(phi_c) * np.cos(r_beam) +
170  np.cos(phi_c) * np.sin(r_beam) * np.cos(theta))
171  lambda_circle = lambda_c + np.arctan2(np.sin(theta) * np.sin(r_beam) * np.cos(phi_c),
172  np.cos(r_beam) - np.sin(phi_c) * np.sin(phi_circle))
173 
174  # 將圓周坐標從弧度轉換回角度，並轉換為 ECEF
175  x_circle, y_circle, z_circle = sph_to_ecef(np.degrees(lambda_circle),
176  np.degrees(phi_circle), R_earth_km)
177  return x_circle, y_circle, z_circle
