738  for (uint32_t i = 0; i < daTimeSlotsConf.size(); i++)
739  {
740  TimeSlotConf* tsConf = daTimeSlotsConf[i];
741  if (tsConf->GetSlotType() == TimeSlotConf::SatTimeSlotType_t::SLOT_TYPE_TRC)
742  {
743  // superframe tx time
744  double sfDuration = m_superframe->GetSuperframeConf()->GetDuration();
745  double frameStartTime = sfCounter*sfDuration;
746 
747  double slotStartTime = tsConf->GetStartTime();
748 
749  uint8_t frameId = std::get<1>(daTimeSlotsInfo);
750  double symbolRate = m_superframe->GetSuperframeConf()->GetFrameConf(frameId)->GetSymbolRateInBauds();
751  double burstDuration = m_superframe->GetSuperframeConf()->GetFrameConf(frameId)->GetWaveformConf()->GetWaveform(tsConf->GetWaveFormId())->GetBurstDuration(symbolRate);
752 
753  double difference = timeReceived - frameStartTime - slotStartTime - burstDuration;
754 
755  if (std::fabs(difference) < differenceClosest)
756  {
757  differenceClosest = std::fabs(difference);
758  timeSlotIndexClosest = i;
759  tbtpIndexClosest = sfCounter;
760  }
761  }
762  }
763  }
764 
765  // check the time correction (the time slot start time - rev time)
766  // send logoff
767  // send CMT
768 
769  if (tbtpIndexClosest == 0) return;
770 
771  auto it = m_tbtps.find(tbtpIndexClosest);
772  auto daTimeSlotsInfo = it->second->GetDaTimeslots(utAddr.GetAddressVal());
773  auto timeslots = std::get<2>(daTimeSlotsInfo).second;
774  TimeSlotConf* tsConf = timeslots[timeSlotIndexClosest];
775 
776  if (tsConf->GetSlotType() == TimeSlotConf::SatTimeSlotType_t::SLOT_TYPE_TRC)
777  {
778  // superframe tx time

781 
782  double slotStartTime = tsConf->GetStartTime();

787
