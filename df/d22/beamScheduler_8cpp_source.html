218 void
219 BeamScheduler::WaitUntilOrNewUt(double deadline)
220 {
221  std::unique_lock<std::mutex> lock(m_pendingUtsAddMutex);
222  double waitSec = deadline - Time::Now();
223  if (waitSec > 0)
224  {
225  // No predicate here on purpose: wait_for's predicate form checks
226  // m_pendingUtsAdd first and returns instantly if it's already
227  // non-empty on entry, rather than only reacting to a fresh
228  // notify_one() during this call. When several UTs are added in a
229  // burst (e.g. a synchronized multi-UT handover), that let this
230  // loop skip real waiting entirely for several iterations in a
231  // row, racing m_superframeCount multiple superframes ahead of
232  // real time -- broadcasting TBTPs for superframes on the new
233  // satellite's channel before the just-handed-over UT had even
234  // physically connected to it, so those TBTPs went unreceived and
235  // left a real multi-superframe service gap once the loop caught
236  // back up. A plain wait_for only wakes early for a notify that
237  // arrives while actually asleep here, so at most one iteration
238  // can run ahead of schedule per real wait.
239  m_newUtCv.wait_for(lock, std::chrono::duration<double>(waitSec));
240  }

691  // Per-superframe snapshot for offline plotting
692  // (python_tools/plot_ts_alloc.py via plot_all_superframes.py):
693  // same 7-field format as Frame::GenerateTimeSlots' own
694  // continuously-accumulating "TsAlloc" file (frame.cpp), but
695  // scoped to just this one superframe and written to its own
696  // file, so each superframe's allocation can be plotted on
697  // its own instead of only ever seeing the whole run mixed
698  // together in one file.
699  //
700  // This whole computation (walking every UT's time slots,
701  // building the per-slot text, tallying per-UT allocated
702  // bytes, and writing both the sf_<N>_TsAlloc snapshot and
703  // each UT's tsAllocKbps line) scales with UT count and time
704  // slots per UT -- exactly the kind of cost that must never
705  // sit in Schedule()'s own critical path, so it's deferred to
706  // the background worker as one job instead of running here.
707  // Safe to defer: tbtp is never freed after TbtpSent() (it's
708  // kept in m_tbtps for the scheduler's lifetime, so the
709  // pointer stays valid), and m_superframe/its FrameConf/
710  // Waveform config objects are set up once at Initialize()
711  // and never mutated per-superframe, so reading them later
712  // sees the same values Schedule() would have seen right now.
713  // m_superframeCount is captured by value (a snapshot of
714  // *this* superframe's index) since the live member keeps
715  // advancing on Schedule()'s own thread. The only
716  // observable tradeoff: the tsAllocKbps line's own Time::Now()
717  // timestamp reflects whenever the background worker gets to
718  // it (typically sub-millisecond later) rather than the exact
719  // instant the TBTP was sent -- acceptable for stats/plotting,
720  // and everything else (each slot's own scheduled start time,
721  // sf_<N>_TsAlloc's content) is unaffected since those come
722  // from tbtp's own fixed data, not from when this runs.
723  uint64_t superframeCount = m_superframeCount;
724  EnqueueBackgroundWork([this, tbtp, superframeCount]() {
725  std::ostringstream sfOss;
726 
727  // unordered_map (hash lookup) instead of map (string-compare tree)
728  // -- purely a lookup-cost improvement, iteration order doesn't
729  // matter below (each UT gets its own separate output file).
730  std::unordered_map<std::string, uint32_t> utRcAlloc;
731  for(const auto& tsConfsMap : tbtp->GetData()->TbtpMessage) // by ref: was copying the whole tuple (incl. its slot vector) per UT
732  {
733  const auto& utAddr = tsConfsMap.first; // by ref: was copying the address string per UT
734 
735  auto frameID = std::get<1>(tsConfsMap.second);
736  const auto& tsConfs = std::get<2>(tsConfsMap.second).second; // by ref: was copying the whole slot vector per UT
737  FrameConf* frameConf = m_superframe->GetSuperframeConf()->GetFrameConf(frameID);
738  int utId = m_nccMac->GetUtId(utAddr);
739  // Frame-constant, hoisted out of the per-slot loop below instead
740  // of being looked up again for every single time slot.
741  double symbolRateInBauds = frameConf->GetSymbolRateInBauds();
742 
743  // Single lookup-or-insert for this UT, reused by every slot below
744  // instead of a fresh map lookup per slot (the loop that used to
745  // matter most once time slot counts grow).
746  uint32_t& allocBytes = utRcAlloc[utAddr];
747 
748  for(auto tsConf : tsConfs)
749  {
750  uint32_t wfID = tsConf->GetWaveFormId();
751  Waveform* waveform = frameConf->GetWaveformConf()->GetWaveform(wfID);
752  uint32_t tsBytes = waveform->GetPayloadInBytes();
753  allocBytes += tsBytes;
754 
755  double wfLenUs = (waveform->GetBurstLengthInSymbols()/symbolRateInBauds)*1e6;
756  sfOss << int(frameID) << " " << tsConf->GetCarrierId() << " "
757  << tsConf->GetStartTime()*1e6 << " " << wfLenUs << " "
758  << wfID << " " << utId << " " << tsBytes << "\n";
759  }
760  }
761 
762  if (!sfOss.str().empty())
763  {
764  m_sfPlotLog->WriteToFile(sfOss.str(), m_log->GetFilePath() + "superframe_plots/",
765  "sf_" + std::to_string(superframeCount) + "_TsAlloc");
766  }
767 
768  std::ostringstream rcOss;
769  for(const auto& rcAlloc : utRcAlloc)
770  {
771  double allocKbps = (rcAlloc.second*8)/m_superframe->GetSuperframeConf()->GetDuration()/1000;
772 
773  rcOss.str("");
774  rcOss.clear();
775  rcOss << std::fixed << std::setprecision(5) << Time::Now() << " " << allocKbps << "\n";
776 
777  std::string fileName = "UT_" + rcAlloc.first + "_tsAllocKbps";
778  m_log->WriteToFile(rcOss.str(),m_nccMac->GetLog()->GetFilePath(),fileName);
779  }
780  });
781  }
782  }
783 
784  m_log->ClearOss(oss);
785  oss << "[Beam scheduler] Scheduling spending time: " << (Time::Now() - tbtp_proc_start_time) << "\n";
786  m_log->WriteToFile(oss.str());
787 
788  m_log->ClearOss(oss);
789  uint64_t current_superframe_index = static_cast<uint64_t>(std::floor(Time::Now()/m_superframe->GetSuperframeConf()->GetDuration()));
790  oss << "Scheduled superframe index: " << m_superframeCount << ", current superframe index: " << current_superframe_index << ", diff: " << (m_superframeCount - current_superframe_index) << "\n";
791  m_log->WriteToFile(oss.str());
792 
793  m_log->ClearOss(oss);
794  oss << "[Beam scheduler] Schedule complete (" << int(schedlCount) << ")\n";
795  m_log->WriteToFile(oss.str());
796 
797  schedlCount++;
798  m_superframeCount++;
799 
800  // std::thread genFigsThread(runPythonScript);
801  // genFigsThread.join();
802 
803  next_schel_start_time += m_superframe->GetSuperframeConf()->GetDuration();
804  double schel_start_d = next_schel_start_time - Time::Now();
805  if(schel_start_d > 0)
806  // WaitUntilOrNewUt(next_schel_start_time);
807  Time::Sleep(schel_start_d);

867 void
868 BeamScheduler::SendCmtMessage(int sock, Address utAddr, double timeReceived)
869 {
870  // Find the closest time slot to the pkt rev time from tbtp in m_tbtps
871  double differenceClosest = 1000000;
872  uint32_t tbtpIndexClosest = 0;
873  uint32_t timeSlotIndexClosest = 0;
874 
875  for (auto pair : m_tbtps)
876  {
877  uint32_t sfCounter = pair.first;
878  TbtpCtrlMsg* tbtp = pair.second;
879  auto daTimeSlotsInfo = tbtp->GetDaTimeslots(utAddr.GetAddressVal());
880  auto daTimeSlotsConf = std::get<2>(daTimeSlotsInfo).second;
881  for (uint32_t i = 0; i < daTimeSlotsConf.size(); i++)
882  {
883  TimeSlotConf* tsConf = daTimeSlotsConf[i];
884  if (tsConf->GetSlotType() == TimeSlotConf::SatTimeSlotType_t::SLOT_TYPE_TRC)
885  {
886  // superframe tx time
887  double sfDuration = m_superframe->GetSuperframeConf()->GetDuration();
888  double frameStartTime = sfCounter*sfDuration;
889 
890  double slotStartTime = tsConf->GetStartTime();
891 
892  uint8_t frameId = std::get<1>(daTimeSlotsInfo);
893  double symbolRate = m_superframe->GetSuperframeConf()->GetFrameConf(frameId)->GetSymbolRateInBauds();
894  double burstDuration = m_superframe->GetSuperframeConf()->GetFrameConf(frameId)->GetWaveformConf()->GetWaveform(tsConf->GetWaveFormId())->GetBurstDuration(symbolRate);
895 
896  double difference = timeReceived - frameStartTime - slotStartTime - burstDuration;
897 
898  if (std::fabs(difference) < differenceClosest)
899  {
900  differenceClosest = std::fabs(difference);
901  timeSlotIndexClosest = i;
902  tbtpIndexClosest = sfCounter;
903  }
904  }
905  }
906  }
907 
908  // check the time correction (the time slot start time - rev time)
909  // send logoff
910  // send CMT
911 
912  if (tbtpIndexClosest == 0) return;
913 
914  auto it = m_tbtps.find(tbtpIndexClosest);
915  auto daTimeSlotsInfo = it->second->GetDaTimeslots(utAddr.GetAddressVal());
916  auto timeslots = std::get<2>(daTimeSlotsInfo).second;
917  TimeSlotConf* tsConf = timeslots[timeSlotIndexClosest];
918 
919  if (tsConf->GetSlotType() == TimeSlotConf::SatTimeSlotType_t::SLOT_TYPE_TRC)
920  {
921  // superframe tx time
922  double sfDuration = m_superframe->GetSuperframeConf()->GetDuration();
923  double frameStartTime = it->second->GetSfCounter()*sfDuration;
924 
925  double slotStartTime = tsConf->GetStartTime();
926 
927  uint8_t frameId = std::get<1>(daTimeSlotsInfo);
928  double symbolRate = m_superframe->GetSuperframeConf()->GetFrameConf(frameId)->GetSymbolRateInBauds();
929  double burstDuration = m_superframe->GetSuperframeConf()->GetFrameConf(frameId)->GetWaveformConf()->GetWaveform(tsConf->GetWaveFormId())->GetBurstDuration(symbolRate);
930 
931  double difference = frameStartTime + slotStartTime + burstDuration - timeReceived;
932 
933  double differenceNcr = difference*std::pow(10,6) * 27;
934 
935  if (differenceNcr > 16256000 || differenceNcr < -16256000)
936  {
937  std::ostringstream oss;
938  oss << "[Beam scheduler] Burst Time Correction outside bounds, should be at least -16256 and at most 16256, but got " << differenceNcr << ". Forcing logoff of UT " << utAddr.GetAddressVal() << std::endl;
939  m_log->WriteToFile(oss.str());
940 
941  // send logoff
942  LogoffCtrlMsg* logoffMsg = new LogoffCtrlMsg();
943  m_netDev->SendControlMsg(sock,utAddr.GetAddressVal(),logoffMsg);
944  m_log->WriteToFile("[Beam scheduler] Send LOGOFF_RESPONSE_CTRL_MSG packet to " + utAddr.GetAddressVal() + " !!\n");
945  // remove UT
946  RemoveUt(utAddr);
947  m_log->WriteToFile("[Beam scheduler] Send LOGOFF_RESPONSE_CTRL_MSG packet to " + utAddr.GetAddressVal() + ", and remove the UT information !!\n");
948  std::cout << "[Beam scheduler] Send LOGOFF_RESPONSE_CTRL_MSG packet to " + utAddr.GetAddressVal() + ", and remove the UT information !!\n";
949  }
950  else
951  {
952  m_log->WriteToFile("[Beam scheduler] Send CMT_CTRL_MSG packet to LLC queue !!\n");
953  }
954 
955  return;
956  }
