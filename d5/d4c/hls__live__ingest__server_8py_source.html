2 """
3 hls_live_ingest_server.py — the uplink-push counterpart to plain
4 `python3 -m http.server`: instead of only serving GETs, this accepts POSTed
5 HLS segments and playlist updates from hls_live_push_client.py and re-serves
6 them over GET, so anyone watching the stream (e.g. ffplay pointed at this
7 server) sees exactly what has actually arrived so far — not what the origin
8 encoder has produced but hasn't reached here over the (possibly congested)
9 link yet.
10 
11 Two POST endpoints:
12  POST /segment/<name> — body is the raw .ts segment bytes, written to
13  <storage_dir>/<name>.
14  POST /playlist — body is the current full .m3u8 playlist text (the
15  push client always sends its complete, current
16  local copy, not a diff), overwriting
17  <storage_dir>/live.m3u8. This is deliberately the
18  *pushed* client's own view of the playlist, so a
19  viewer here only ever sees segments that have
20  actually finished uploading.
21 
22 GET requests are served as plain static files out of <storage_dir> (same
23 convention as `python3 -m http.server`), so a client can fetch
24 live.m3u8/<segment>.ts exactly like any other HLS origin server.
25 
26 One instance of this server is started per UE (see launch_uts_and_ues),
27 each on its own port and storage directory, rather than one shared
28 multi-tenant server for all UEs.
29 
30 Usage:
31  hls_live_ingest_server.py <port> <storage_dir>
32 """

39 class IngestHandler(SimpleHTTPRequestHandler):
40  def _write_atomic(self, dest, body):
41  """Write body to dest via a temp file + os.replace(), so a GET
42  handled by a different thread of this ThreadingHTTPServer can never
43  observe a partially-written file — a plain open(dest,"wb") truncates
44  dest to 0 bytes before the write completes, and live.m3u8 in
45  particular gets overwritten often enough (every successful push)
46  while being polled often enough (every ffplay playlist reload) that
47  a concurrent GET landing in that window would see a truncated/torn
48  file. os.replace() is an atomic rename on POSIX, so any GET sees
49  either the fully-old or fully-new file, never a partial one."""
50  tmp_dest = dest + ".tmp"
51  with open(tmp_dest, "wb") as f:
52  f.write(body)
53  os.replace(tmp_dest, dest)
