1 #!/usr/bin/env python3
2 """
3 hls_live_client.py — simulates a real HLS *live* stream viewer: polls a
4 growing HLS playlist and downloads new segments as they appear, rather than
5 assuming the whole segment list is known upfront the way a VOD-style
6 playlist does. If the network can't keep up with the live encoder's pace,
7 this client simply falls further and further behind the live edge — exactly
8 like a real player would — instead of failing or catching up by skipping
9 ahead.
10 
11 Each poll of the playlist:
12  - discovers any segment names not already downloaded, and fetches them
13  (each a separate, independent curl invocation with its own timeout
14  budget — a per-segment model, not web_page_client.py's --next-chained
15  groups, since here segments arrive one at a time over real time rather
16  than all being known at once).
17  - if nothing new appeared and the playlist doesn't yet contain
18  #EXT-X-ENDLIST, the live encoder just hasn't produced the next segment
19  yet — wait POLL_INTERVAL_S and check again.
20  - if nothing new appeared and #EXT-X-ENDLIST is present, the live encoder
21  has genuinely finished and every known segment has been fetched — done.
22 
23 Runs until the stream ends or <duration_s> elapses, whichever comes first.
24 On exit, writes a small key=value summary to <summary_path>.
25 
26 LIVE LATENCY: each fetched segment's log line reports how far behind the
27 live edge it was the moment the download finished — i.e. real streaming
28 latency, not just download time. This is computed from the segment file's
29 Last-Modified response header (Python's http.server sets this from the
30 file's actual mtime on NCC's disk, which — since the encoder writes each
31 segment at its own real playback pace via ffmpeg -re — is effectively "when
32 this segment was produced"): latency = (time download finished) - (segment's
33 Last-Modified time). Printed and tracked in real time as segments arrive, so
34 watching this process's log shows the lag trend live as the run progresses.
35 
36 usage:
37  hls_live_client.py <playlist_url> <duration_s> <summary_path>
38  e.g. hls_live_client.py http://10.0.255.254:5101/live.m3u8 300 \\
39  log_file/UE_10.0.200.101/video_live_duration.txt
40 """

48 CONNECT_TIMEOUT_S = 60 # see web_page_client.py — measured ut_queueing_delay
49 # averages ~15.6s and peaks ~70s under congestion, so a short connect-timeout

73 def fetch_text(url):
74  """GET url via curl and return the response body as text, or "" if curl
75  failed OR the body doesn't actually look like an HLS playlist. Before the
76  live encoder has created the playlist file yet, the server returns a 404
77  with an HTML error body — that body is non-empty, so without checking
78  both the exit code and the #EXTM3U marker, it would be misread as a
79  valid (if oddly formatted) playlist and every line of the error page
80  would be "discovered" as a bogus segment name."""
81  result = subprocess.run(
82  ["curl", "-s", "--connect-timeout", str(CONNECT_TIMEOUT_S), url],
83  capture_output=True, text=True, timeout=CONNECT_TIMEOUT_S + 10,
84  )
85  if result.returncode != 0:
86  return ""
87  if "#EXTM3U" not in result.stdout:
88  return ""
89  return result.stdout

92 def fetch_segment(seg_url, timeout_s):
93  """GET seg_url via curl, discarding the body but keeping the response
94  headers (-D -) to read Last-Modified. Returns (elapsed_s, success,
95  produced_at_unix_or_None) — produced_at is the segment's Last-Modified
96  time (Python's http.server sets it from the file's real mtime on NCC's
97  disk) converted to a Unix timestamp, used to compute live latency in
98  main(); None if the header was missing/unparseable.
99 
100  --fail makes curl treat a non-2xx HTTP response as a failure (its exit
101  code alone doesn't — e.g. a 404 still exits 0 by default). A real stall
102  (sustained near-zero throughput) or a hard hang past timeout_s also
103  counts as a failure, same reasoning as web_page_client.py's
104  fetch_group/fetch_discard."""
105  t0 = time.time()
106  try:
107  result = subprocess.run(
108  ["curl", "-s", "-f", "-D", "-", "--connect-timeout", str(CONNECT_TIMEOUT_S),
109  "--speed-limit", str(STALL_SPEED_BYTES_PER_SEC), "--speed-time", str(STALL_WINDOW_S),
110  seg_url, "-o", "/dev/null"],
111  capture_output=True, text=True, timeout=timeout_s,
112  )
113  except subprocess.TimeoutExpired:
114  return time.time() - t0, False, None
115 
116  elapsed = time.time() - t0
117  if result.returncode != 0:
118  return elapsed, False, None
119 
120  produced_at = None
121  for line in result.stdout.splitlines():
122  if line.lower().startswith("last-modified:"):
123  try:
124  produced_at = email.utils.parsedate_to_datetime(line.split(":", 1)[1].strip()).timestamp()
125  except (ValueError, TypeError):
126  pass
127  break
128  return elapsed, True, produced_at
