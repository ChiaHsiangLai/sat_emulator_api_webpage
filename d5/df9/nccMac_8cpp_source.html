184  delete pkt;

186  }

959 void
960 NccMac::PerformUtHandover(std::string old_utIp, std::string target_sat_ip)
961 {
962  HandoverMsg* utHoMsg = new HandoverMsg();
963  utHoMsg->SetHoMsg(target_sat_ip);
964  // m_netDev->SendControlMsg(1, old_utIp, utHoMsg);
965  SocketManager::SockMap_t sockMap = m_gwSockMap->GetSockMap();
966  std::string MacDestAddrVal = sockMap.begin()->first;
967  int sock = *(sockMap.begin()->second);
968  SendHoPktToGw(sock, old_utIp, MacDestAddrVal, utHoMsg);
969 
970  int sat_id = GetSatId(target_sat_ip);
971  std::string new_utIp = fixIp(old_utIp, sat_id);
972 
973  auto ut_rt = m_utRoutingTable->GetRoutingTable();
974 
975  Address old_utAddr(old_utIp, AddressType::MAC48, old_utIp.size());
976  BeamScheduler* oldBeamSchlr = m_nccCore->GetBeamScheduler(ut_rt[old_utIp].satId, ut_rt[old_utIp].beamId);
977 
978  // *** [HANDOVER RBDC CARRY-OVER] ***
979  // Snapshot the UT's in-flight RBDC/VBDC demand from the old beam scheduler's
980  // Dama before it is discarded, so the new beam scheduler can seed its Dama
981  // with it and avoid granting CRA-only until the next CR arrives.
982  DamaState damaState = oldBeamSchlr->GetDama(old_utAddr)->GetState();
983 
984  // update routing table
985  UtRoutingInfo rt_info;
986  rt_info = ut_rt[old_utIp];
987  rt_info.satIp = target_sat_ip;
988  rt_info.satId = sat_id;
989  m_utRoutingTable->SetRouting(new_utIp,rt_info);
990  m_utRoutingTable->RemoveRouting(old_utIp);
991 
992  // inform new beam scheduler add UT info
993  Address new_utAddr(new_utIp, AddressType::MAC48, new_utIp.size());
994  BeamScheduler* beamSchlr = m_nccCore->GetBeamScheduler(sat_id,1);
995  if(beamSchlr)
996  {
997  if(!beamSchlr->HasUT(new_utAddr))
998  {
999  m_nccCore->AddUt(1,new_utAddr,sat_id,1,&damaState);
1000  }
1001  }
1002 
1003  // update handover state
1004  int utId = GetUtId(new_utIp);
1005  std::string ut_dev = "UT" + std::to_string(utId);
1006  // m_isDevHo[ut_dev] = true;
1007 
1008  // *** [MAKE-BEFORE-BREAK HANDOVER] ***
1009  // Don't cut the UT from the old beam scheduler the instant the new one
1010  // is told about it. The physical link switch (UT reacting to the
1011  // handover message it's about to receive) is empirically much faster
1012  // (~0.25s observed) than the new scheduler's structural minimum TBTP
1013  // pipeline-fill time (>=1 superframe -- see StartSchedlThread's
1014  // schedl_start_d, NCC.cpp), so an immediate RemoveUt() here left the
1015  // UT briefly served by neither scheduler right around the physical
1016  // switch -- confirmed empirically: one whole superframe's TBTP
1017  // silently missing from the UT's own log right at the handover
1018  // instant, even with RA enabled as a fallback. Keep the old scheduler
1019  // granting it for one more superframe -- matching UtMAC's own
1020  // symmetric delay (utMac.cpp, CheckLinkStatusAndCnt) before it
1021  // actually executes the physical switch -- so there's overlap
1022  // instead of a gap.
1023  double holdoverDelay = oldBeamSchlr->GetSuperframe()->GetSuperframeConf()->GetDuration();
1024  std::thread([oldBeamSchlr, old_utAddr, holdoverDelay]() {
1025  Time::Sleep(holdoverDelay);
1026  oldBeamSchlr->RemoveUt(old_utAddr);
1027  }).detach();
1028 }
