959 {

969  std::string new_utIp = fixIp(old_utIp, sat_id);
970 
971  auto ut_rt = m_utRoutingTable->GetRoutingTable();
972 
973  Address old_utAddr(old_utIp, AddressType::MAC48, old_utIp.size());
974  BeamScheduler* oldBeamSchlr = m_nccCore->GetBeamScheduler(ut_rt[old_utIp].satId, ut_rt[old_utIp].beamId);
975 
976  // *** [HANDOVER RBDC CARRY-OVER] ***
977  // Snapshot the UT's in-flight RBDC/VBDC demand from the old beam scheduler's
978  // Dama before it is discarded, so the new beam scheduler can seed its Dama
979  // with it and avoid granting CRA-only until the next CR arrives.

986  rt_info.satId = sat_id;

988  m_utRoutingTable->RemoveRouting(old_utIp);
989 
990  // inform new beam scheduler add UT info
991  Address new_utAddr(new_utIp, AddressType::MAC48, new_utIp.size());
992  BeamScheduler* beamSchlr = m_nccCore->GetBeamScheduler(sat_id,1);
993  if(beamSchlr)

995  if(!beamSchlr->HasUT(new_utAddr))
996  {
997  m_nccCore->AddUt(1,new_utAddr,sat_id,1,&damaState);
998  }
999  }
1000 
1001  // update handover state
1002  int utId = GetUtId(new_utIp);
1003  std::string ut_dev = "UT" + std::to_string(utId);
1004  // m_isDevHo[ut_dev] = true;

1006  oldBeamSchlr->RemoveUt(old_utAddr);
1007 }
