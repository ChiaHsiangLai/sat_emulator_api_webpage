107 def launch_wireshark(net, intf=None):
108  """Open a Wireshark window, pre-selected (and capturing, via -k) on intf
109  if given, so the caller doesn't have to pick and start it manually.
110 
111  Deliberately no explicit -w here. Passing one (pointed at a plain
112  world-writable directory, even a pre-created 0666 file, even though
113  this whole process runs as root) still hit "Permission denied" opening
114  that *output* file specifically — opening the *capture device* itself
115  always worked fine, so this is something in dumpcap's own privilege-
116  separation handling around -w, not an ordinary DAC permission gap.
117  Left unset, dumpcap falls back to its own default temp file under
118  /tmp/wireshark_<intf><random>.pcapng, which empirically always works
119  (confirmed: several such files from earlier sessions, all valid,
120  non-empty pcapng). save_wireshark_capture() copies that temp file into
121  a permanent home under WIRESHARK_CAPTURE_DIR once the simulation ends.
122 
123  Mininet Host nodes (UE, UT, SAT, GW, NCC — anything from addHost) each get
124  their own network namespace (Node's inNamespace=True default, never
125  overridden by Host); switches default to inNamespace=False, i.e. root
126  namespace (Mininet.__init__'s inNamespace=False propagates into
127  addSwitch's params). So an interface belonging to a Host (e.g. a UE's own
128  "ue101-eth0") is invisible to a plain root-namespace process — Wireshark
129  launched as a plain subprocess wouldn't even list it, let alone capture
130  it — it must be launched *inside* that specific node via node.cmd()
131  instead. A switch port (e.g. "sw1-eth1") has no such namespace to enter,
132  so node.cmd() on it behaves the same as a plain subprocess anyway.
133 
134  intf's owning node name is derived from mininet's own naming convention
135  (node_name + "-eth" + N, e.g. "ue101-eth0" -> "ue101") — this is the same
136  convention every addLink() call in this script's topology already uses,
137  not something layered on top here.
138  """
139  if not intf:
140  # No specific interface: same as before, a plain subprocess in the
141  # root namespace — only sees switch ports, not any Host's own intf.
142  subprocess.Popen(['wireshark'])
143  return
144  node_name = intf.split('-eth')[0]
145  node = net.get(node_name)
146  if node is None:
147  print(f"[launch_wireshark] No node named '{node_name}' (parsed from "
148  f"interface '{intf}') — check WIRESHARK_INTF.")

177 def wait_for_ut_logon(ut_ip, timeout_s=UT_LOGON_TIMEOUT_S, poll_interval_s=0.5):
178  """Block until UT ut_ip's own MacLogFile shows it received a confirmed
179  logon from the satellite, or until timeout_s elapses.
180 
181  The UT process logs "send logon success !!" the moment it transmits its
182  own logon request — that only means the request was sent, not that NCC
183  ever granted it. The actual confirmation is a separate line printed only
184  after the satellite responds: "Received response from Sat (...): Logon
185  success !! RA channel ID: <n>" — "RA channel ID" only ever appears in
186  that confirmation line, so it's used here as the unambiguous marker
187  rather than "logon success" (which would also match the UT's own
188  unconfirmed send).
189 
190  Returns True once logon is confirmed, False if timeout_s elapses first
191  (e.g. a misconfigured BeamScheduler/Frame0/1/2LogonEnable leaves the UT
192  stuck retrying logon forever — see validate_config's beam-scheduler
193  check for the same failure mode). Callers should treat False as "proceed
194  anyway, but warn" rather than hang the whole launch indefinitely.
195  """
196  mac_log_path = os.path.join(LOG_DIR, f"UT_{ut_ip}", "MacLogFile")
197  deadline = time.time() + timeout_s
198  while time.time() < deadline:
199  if os.path.exists(mac_log_path):
200  with open(mac_log_path) as f:
201  if "RA channel ID" in f.read():
202  return True
203  time.sleep(poll_interval_s)
204  return False

207 def ffplay_watch_cmd(playlist_url, window_title, retry_log_path):
208  """Build a shell command that waits for playlist_url to become reachable,
209  then plays it with ffplay, retrying the whole ffplay invocation if it
210  fails to actually open the stream.
211 
212  ffplay exits 0 even when it never managed to play anything (e.g. a
213  connect timeout, or a 404 partway through a live playlist) — its exit
214  code alone can't tell success from failure, so this greps ffplay's own
215  stderr for either of two known-fatal patterns:
216  - "<url>: <error>", the line ffmpeg prints whenever avformat_open_input
217  fails outright (confirmed empirically for connection-timeout/
218  refused/4xx cases opening the *playlist* itself), or
219  - "Failed to open segment", printed when a *specific segment* the
220  playlist lists 404s (ffplay does not retry or recover from this on
221  its own — it just sits there printing "nan" progress forever).
222  Neither line appears during a genuinely successful playback, so their
223  absence is what ends the retry loop. Uses FFPLAY_TIMEOUT_US (see its own
224  comment) as ffplay's own protocol-level connect/IO timeout, which is
225  otherwise far shorter than what our own scripts tolerate under the same
226  satellite congestion.
227 
228  retry_log_path must be unique per concurrently-running ffplay watcher
229  (e.g. include the UE name/number) to avoid two instances clobbering the
230  same temp file.
231  """
232  return (
233  f"until curl -sf -o /dev/null {playlist_url}; do sleep 1; done; "
234  f"while true; do "
235  f'ffplay -timeout {FFPLAY_TIMEOUT_US} -autoexit -window_title "{window_title}" {playlist_url} '
236  f"2>&1 | tee {retry_log_path}; "
237  f'(grep -qF "{playlist_url}:" {retry_log_path} || grep -qF "Failed to open segment" {retry_log_path}) || break; '
238  f"sleep 2; "
239  f"done"
240  )

298 def ffplay_listen_rtmp_cmd(port, window_title, duration_s):
299  """Build a shell command that runs ffplay as a one-shot RTMP server:
300  it accepts a single incoming push connection and plays it live as
301  frames arrive, no separate ingest/relay process and no local file
302  involved (verified empirically: ffmpeg/ffplay's rtmp protocol supports
303  -listen 1, and a pushed stream decodes/plays correctly through it).
304 
305  RTMP's listen mode serves exactly one connection per invocation and
306  exits once that connection ends (whether the sender finished normally
307  or dropped), so this loops for up to duration_s seconds so the window
308  keeps waiting for a(nother) push instead of just closing after the
309  first session — e.g. if the sender process gets restarted mid-run.
310 
311  -fflags nobuffer -flags low_delay trim ffplay's own internal
312  buffering/decode latency; left at ffplay's default probesize/
313  analyzeduration rather than shrinking those too, since RTMP already
314  carries codec params in its onMetaData (so probing is cheap here) and
315  pushing probesize too low risks ffplay failing to pick up the audio
316  stream at all."""
317  return (
318  f"end=$(( $(date +%s) + {duration_s} )); "
319  f"while [ $(date +%s) -lt $end ]; do "
320  f'ffplay -fflags nobuffer -flags low_delay -listen 1 -window_title "{window_title}" rtmp://0.0.0.0:{port}/live; '
321  f"done"
322  )

666 def dynamic_link_control(net, topo, topo_util, sat_ip, sat_pos_file,
667  get_sat_pos_from_tle, init_time, interval=1.0,
668  dat_queue=None, link_configure_enabled=True):
669  sw_name = f"sw{sat_ip.split('.')[-2]}"
670  sw_node = net.get(sw_name)
671  host_states = {}
672  host_channel = {}
673  host_locations = {}
674 
675  ELEVATION_THRESHOLD = 0
676  step_time = 1
677  start_time = START_TIME
678  current_time = init_time

680  def apply_netem(node_name, intf_name, delay_ms, jitter_ms, loss_pct):
681  """Apply netem qdisc settings on a single interface.
682 
683  Serialised by _netem_lock: multiple link-control threads call this
684  on shared nodes (gw1, ut1, …) concurrently, which would corrupt
685  Mininet's bash pipe/sentinel if not locked.
686  """
687  node = net.get(node_name)
688  with _netem_lock:
689  node.cmd(
690  f"tc qdisc replace dev {intf_name} root netem "
691  f"delay {delay_ms:.2f}ms {jitter_ms:.2f}ms "
692  f"loss {loss_pct:.4f}%"
693  )
694 
695  while not stop_event.is_set():
696  try:
697  if get_sat_pos_from_tle:
698  sat_lat, sat_lon, sat_h = topo_util.get_sat_pos(sat_ip, sat_pos_file, current_time)
699  else:
700  sat_lat, sat_lon, sat_h = topo_util.get_sat_pos_from_file(sat_ip, sat_pos_file)
701 
702  current_uts_info = []
703 
704  for link in net.links:
705  if link.intf1.node != sw_node and link.intf2.node != sw_node:
706  continue
707 
708  if link.intf1.node == sw_node:
709  sw_intf, host_intf, host_node = link.intf1, link.intf2, link.intf2.node
710  else:
711  sw_intf, host_intf, host_node = link.intf2, link.intf1, link.intf1.node
712 
713  h_name = host_node.name.lower()
714  if h_name.startswith('sat'):
715  continue
716 
717  # ── Initialize per-host state tracking ───────────────────────
718  if h_name not in host_states:
719  host_states[h_name] = {'last_loss': -1.0, 'last_delay': -1.0}
720  state = host_states[h_name]
721 
722  # ── Determine mobility filename from host role ─────────────
723  node_path_fileName = None
724  node_ip = None
725  if h_name.startswith("gw"):
726  gwId = int(h_name[2:])
727  node_ip = f"10.0.255.{gwId}"
728  node_path_fileName = f"GW_{node_ip}_mobility"
729  elif h_name.startswith("ut"):
730  utId = int(h_name[2:])
731  node_ip = f"10.0.1.{utId + 100}"
732  node_path_fileName = f"UT_{node_ip}_mobility"
733 
734  if not node_path_fileName:
735  continue
736 
737  try:
738  src_file = os.path.join(DATA_DIR, "mobility_files", node_path_fileName)
739  if h_name.startswith("gw"):
740  h_lat, h_lon, h_h = topo_util.get_gw_pos_from_file(node_ip, src_file)
741  else:
742  h_lat, h_lon, h_h = topo_util.get_ut_pos_from_file(node_ip, src_file)
743 
744  h_time = time.time()
745  h_speed = 0.0
746  elevation = topo_util.calculate_elevation(
747  h_lat, h_lon, h_h, sat_lat, sat_lon, sat_h
748  )
749 
750  if h_name not in host_channel:
751  host_channel[h_name] = ChannelModel(
752  FadingModelType='markov', elevation=elevation
753  )
754  host_locations[h_name] = (h_time, h_lat, h_lon, h_h, h_speed)
755 
756  # ── Compute link loss / delay from elevation ───────────
757  if elevation < ELEVATION_THRESHOLD:
758  # Satellite below horizon — full loss
759  loss_pct = 100.0
760  delay_ms = 0.0
761  jitter_ms = 0.0
762  fading_dB = float('nan')
763  snr_db = float('nan')
764  else:
765  distance = topo_util.get_distance(
766  h_lat, h_lon, h_h, sat_lat, sat_lon, sat_h
767  )
768  delay_ms = (distance * 1e3 / 3e8) * 1000.0
769  cm = host_channel[h_name]
770  fading_dB, snr_db = cm.GetSNR(
771  topo_util, h_lat, h_lon, h_h,
772  sat_lat, sat_lon, sat_h, h_speed, elevation
773  )
774  loss_pct = cm.Get_pkt_loss_ratio(snr_db)
775  jitter_ms = 0.0
776 
777  # ── Apply TC only when metrics meaningfully change ─────
778  loss_changed = abs(loss_pct - state['last_loss']) > 0.01
779  delay_changed = abs(delay_ms - state['last_delay']) > 0.1
780 
781  if link_configure_enabled and (loss_changed or delay_changed):
782  apply_netem(h_name, host_intf.name, delay_ms, jitter_ms, loss_pct)
783  apply_netem(sw_name, sw_intf.name, delay_ms, jitter_ms, loss_pct)
784  state['last_loss'] = loss_pct
785  state['last_delay'] = delay_ms
786 
787  # Log link state change
788  status_str = ("DOWN(loss=100%)" if loss_pct >= 100.0
789  else f"UP(loss={loss_pct:.2f}%)")
790  info(f"[Link] {h_name} <-> Sat_{sat_ip}: {status_str} "
791  f"delay={delay_ms:.1f}ms elev={elevation:.2f}°\n")
792 
793  # ── Collect UT info for visualizer + elevation log ─────
794  if h_name.startswith("ut"):
795  # Which satellite this UT is *actually* attached to
796  # right now (post-handover), as opposed to sat_ip --
797  # the satellite whose elevation/loss this particular
798  # iteration happens to be computing. The two differ
799  # whenever a UT is visible to (and being netem'd
800  # against) a satellite it isn't currently routed
801  # through.
802  attached_sat_id = get_ut_attached_sat_id(net, h_name)
803  attached_sat = f"10.0.{attached_sat_id}.1" if attached_sat_id is not None else None
804 
805  current_uts_info.append({
806  'id': h_name,
807  'lat': h_lat,
808  'lon': h_lon,
809  'alt': h_h,
810  'elev': elevation,
811  'status': 'down' if loss_pct >= 100.0 else 'up',
812  'attached_sat': attached_sat,
813  })
814 
815  filepath = os.path.join(LOG_DIR, f"Sat_{sat_ip}_{h_name}_elevAngle")
816  with open(filepath, "a") as f:
817  now = time.time()
818  f.write(
819  f"{now - start_time:.4f} {elevation:.4f} "
820  f"{fading_dB:.4f} {snr_db:.4f} {h_speed:.4f}\n"
821  )
822 
823  except Exception:
824  pass
825 
826  if dat_queue:
827  dat_queue.put({
828  'sat_id': sat_ip,
829  'lat': sat_lat,
830  'lon': sat_lon,
831  'alt': sat_h,
832  'uts_info': current_uts_info,
833  })
834 
835  except Exception as e:
836  print(f"Loop Error for Sat {sat_ip}: {e}")

839  time.sleep(interval)
840

870 def setup_pure_routing(net, topo):
871  """Install static IP routes so all nodes can reach each other.
872 
873  This function only configures Linux kernel routing; no SNS3 satellite
874  protocol is involved here.
875  - GW1: IP forwarding enabled to relay packets between the satellite
876  subnet (10.0.{sat_id}.0/24) and the NCC feeder subnet (10.0.255.0/24).
877  - NCC: one /24 route per satellite's user-link subnet, pointing via GW1,
878  so NCC can route downlink packets back to UTs.
879  - UT: default route pointing to GW1's IP on the satellite subnet
880  (10.0.{sat_id}.2). This route is only used by UT's own signaling
881  packets (logon, CR, TBTP); UE data packets are intercepted by policy
882  routing in setup_tun_tunnel before reaching this default route.
883  """
884  gw1 = net.get('gw1')
885  ncc = net.get('ncc')
886 
887  # GW1 must forward packets between the satellite subnet and the NCC
888  # feeder subnet (10.0.255.0/24).
889  gw1.cmd('sysctl -w net.ipv4.ip_forward=1')
890 
891  # NCC: route each satellite's user-link subnet via GW1 (10.0.255.1)
892  # so that downlink packets (NCC → UT) can reach the satellite subnet.
893  for sat_name in topo.sats:
894  sat_id = topo.sat_to_Id[sat_name]
895  ncc.cmd(f'ip route add 10.0.{sat_id}.0/24 via 10.0.255.1 dev ncc-eth0')
896 
897  # UT: default route via GW1's IP on the satellite subnet (always .2,
898  # since GW index=1 gives offset +1, i.e. 10.0.{sat_id}.{1+1}).
899  # Used only by UT's own signaling; UE data is redirected to tun0
900  # by the policy routing rules set in setup_tun_tunnel.
901  for ut_name in topo.uts:
902  ut = net.get(ut_name)
903  sat_id = topo.sat_to_Id[topo.ut_to_sat[ut_name]]
904  intf = f"{ut_name}-eth{sat_id - 1}"
905  ut.cmd(f'ip route add default via 10.0.{sat_id}.2 dev {intf}')

908 def setup_tun_tunnel(net, topo):
909  """Create tun0 virtual NICs and configure policy routing to steer UE
910  uplink traffic through the SNS3 satellite protocol stack.
911 
912  WHY tun0 IS NEEDED
913  SNS3 is a userspace process and has no ability to intercept packets that
914  the Linux kernel is forwarding. Without tun0, UE packets would be
915  forwarded directly by the kernel via the satellite interface, bypassing
916  the entire SNS3 MAC/PHY scheduling (TBTP, CR, segmentation).
917  tun0 acts as a pipe between the kernel and SNS3:
918  - UT side : kernel routes the packet to tun0; SNS3 calls read() to pick
919  it up and feed it into the satellite protocol stack.
920  - NCC side: SNS3 reassembles segments and calls write() to inject the
921  reconstructed IP packet back into the kernel, which delivers
922  it to the iperf server via normal TCP/UDP demux.
923 
924  UPLINK PATH (UE → NCC, through SNS3)
925  When a UE packet arrives at the UT, the kernel performs IP forwarding
926  (not TCP processing, since the destination is NCC, not the UT itself).
927  iptables marks packets from the UE subnet (10.0.200.0/24) destined for
928  the NCC feeder subnet (10.0.255.0/24) with fwmark=42 in PREROUTING,
929  before the kernel makes its routing decision. Policy routing then
930  redirects marked packets to table 200, which routes them out via tun0
931  instead of the default satellite interface.
932  SNS3 AppInterface reads the full IP packet from tun0, segments it into
933  fixed-size chunks, and transmits each segment in the TBTP-allocated slot
934  via the satellite socket. On the NCC side, SNS3 reassembles the segments
935  and write()s the original IP packet to NCC's tun0; the kernel delivers it
936  to the iperf server.
937 
938  WHY PREROUTING, NOT FORWARD
939  PREROUTING runs before the routing decision, so the fwmark is in place
940  when policy routing is evaluated. Marking in FORWARD (post-routing) is
941  too late: the packet is already committed to the satellite interface.
942 
943  UT SIGNALING PACKETS ARE NOT AFFECTED
944  Packets generated by the SNS3 UT process itself (logon, TBTP, CR) enter
945  the kernel via the OUTPUT chain, which does not pass through PREROUTING.
946  Therefore fwmark=42 is never applied to them, and they follow the main
947  table default route directly to the satellite interface.
948 
949  DOWNLINK PATH (NCC → UE, bypasses SNS3)
950  TCP ACKs and all downlink packets follow pure kernel IP forwarding:
951  NCC → GW → SAT → UT → UE. netem-emulated delay/loss still applies,
952  but TBTP scheduling is not involved in the downlink direction.
953  """
954  gw1 = net.get('gw1')
955  ncc = net.get('ncc')
956 
957  # ── UT side: create tun0 and configure policy routing ────────────────────
958  for ut_name in topo.uts:
959  ut = net.get(ut_name)
960  sat_id = topo.sat_to_Id[topo.ut_to_sat[ut_name]]
961 
962  # Create a layer-3 TUN device; SNS3 AppInterface binds it via
963  # open("/dev/net/tun") + TUNSETIFF ioctl at startup.
964  ut.cmd('ip tuntap add dev tun0 mode tun')
965  ut.cmd('ip link set tun0 up')
966 
967  # Mark UE→NCC packets with fwmark=42 in PREROUTING so that policy
968  # routing can redirect them before the kernel's default forwarding
