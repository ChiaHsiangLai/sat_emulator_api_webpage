27 def draw_topology(net):
28  """
29  Generates and saves a visual graph of the current Mininet topology.
30  Nodes represent hosts and switches; edges represent links annotated
31  with interface names and IP addresses on both ends.
32  The graph is saved to 'topology_graph.png'.
33  """
34  info('*** Generating Topology Graph...\n')
35  G = nx.Graph()
36 
37  # Iterate over all hosts and switches to populate the graph
38  for host in net.hosts + net.switches:
39  G.add_node(host.name)
40  for intf in host.intfList():
41  if intf.link:
42  # Retrieve the two nodes connected by this link
43  node1, node2 = intf.link.intf1.node, intf.link.intf2.node
44 
45  # Get interface name and IP address for each endpoint;
46  # fall back to "No IP" if no address is assigned yet
47  addr1 = intf.link.intf1.ip if intf.link.intf1.ip else "No IP"
48  name1 = intf.link.intf1.name
49  addr2 = intf.link.intf2.ip if intf.link.intf2.ip else "No IP"
50  name2 = intf.link.intf2.name
51 
52  # Build an edge label showing both interface names and their IPs
53  label = f"{name1}({addr1}) <-> {name2}({addr2})"
54  G.add_edge(node1.name, node2.name, label=label)
55 
56  # Use spring layout for automatic node positioning
57  pos = nx.spring_layout(G)
58  plt.figure(figsize=(12, 8))
59 
60  # Draw nodes and edges with labels
61  nx.draw(G, pos, with_labels=True, node_size=2000,
62  node_color='skyblue', font_size=10, font_weight='bold')
63 
64  # Overlay edge labels showing interface names and IP addresses
65  edge_labels = nx.get_edge_attributes(G, 'label')
66  nx.draw_networkx_edge_labels(G, pos, edge_labels=edge_labels, font_size=7)
67 
68  plt.title("Mininet Topology: UT(s) -> SAT1 -> GW -> NCC")
69  plt.savefig("topology_graph.png")

75 def myNetwork(num_uts=1):
76  """
77  Builds and starts the Mininet network.
78 
79  Parameters
80  ----------
81  num_uts : int
82  Number of User Terminal (UT) hosts to create (default: 1).
83  Each UT is assigned a unique IP in the 10.0.1.x subnet,
84  starting from 10.0.1.10, and is connected to s1.
85 
86  Topology overview:
87  - Hosts: ut0, ut1, ..., ut(N-1), sat1, gw, ncc
88  - Switches: s1 (left side, connects all UTs / SAT1 / GW)
89  s2 (right side, connects GW and NCC)
90  - 3 subnets enforce the routing path:
91  10.0.1.x -> UT(s) <-> SAT1
92  10.0.2.x -> SAT1 <-> GW
93  10.0.3.x -> GW <-> NCC
94 
95  After starting the network, secondary IPs and static routes are
96  configured so that ALL traffic follows:
97  UT(s) -> SAT1 -> GW -> NCC (and the symmetric reverse path)
98  """
99 
100  # Use TCLink so bandwidth limits can be applied to each link
101  net = Mininet(topo=None, build=False, link=TCLink)
102 
103  info('*** Adding controller\n')
104  net.addController(name='c0')
105 
106  info('*** Add switches\n')
107  s1 = net.addSwitch('s1') # Primary switch: connects all UTs, SAT1, and GW
108  s2 = net.addSwitch('s2') # Secondary switch: connects GW and NCC
109 
110  # ----------------------------------------------------------------
111  # Create UT hosts dynamically based on num_uts.
112  # IPs start at 10.0.1.10 to avoid collisions with SAT1 (10.0.1.2).
113  # Host names follow the pattern: ut0, ut1, ut2, ...
114  # ----------------------------------------------------------------
115  info(f'*** Adding {num_uts} UT host(s)\n')
116  uts = []
117  for i in range(num_uts):
118  ip = f'10.0.1.{10 + i}/24' # e.g. 10.0.1.10, 10.0.1.11, ...
119  ut = net.addHost(f'ut{i}', ip=ip) # Name: ut0, ut1, ut2, ...
120  uts.append(ut)
121 
122  info('*** Add infrastructure hosts\n')
123  sat1 = net.addHost('sat1', ip='10.0.1.2/24') # Satellite node (bridges subnets 1 and 2)
124  gw = net.addHost('gw', ip='10.0.2.2/24') # Gateway (bridges subnets 2 and 3)
125  ncc = net.addHost('ncc', ip='10.0.3.2/24') # Network Control Center
126 
127  info('*** Add links\n')
128  # Connect every UT to s1
129  for ut in uts:
130  net.addLink(ut, s1, bw=100) # 100 Mbps link: UTx <-> s1
131 
132  # Connect infrastructure nodes to their respective switches
133  net.addLink(sat1, s1, bw=100) # 100 Mbps link: SAT1 <-> s1
134  net.addLink(gw, s1, bw=100) # 100 Mbps link: GW <-> s1
135  net.addLink(gw, s2, bw=100) # 100 Mbps link: GW <-> s2
136  net.addLink(s2, ncc, bw=100) # 100 Mbps link: s2 <-> NCC
137 
138  info('*** Starting network\n')
139  net.build()
140  draw_topology(net) # Capture and save the topology graph before routing is applied
141  net.start()
142 
143  info('*** Configuring Multi-IP and Routes for redirection\n')
144 
145  # ----------------------------------------------------------------
146  # Step 1: Assign secondary IP addresses to SAT1 and GW
147  # so each node can communicate on two different subnets.
148  # ----------------------------------------------------------------
149 
150  # SAT1: add alias 10.0.2.1 so it can talk to GW on the 10.0.2.x subnet
151  sat1.cmd('ifconfig sat1-eth0:1 10.0.2.1 netmask 255.255.255.0')
152  # Enable IP forwarding on SAT1 so it can route packets between subnets
153  sat1.cmd('sysctl -w net.ipv4.ip_forward=1')
154

156  gw.cmd('ifconfig gw-eth1 10.0.3.1 netmask 255.255.255.0')
157  # Enable IP forwarding on GW so it can route packets between subnets
158  gw.cmd('sysctl -w net.ipv4.ip_forward=1')
159 
160  # ----------------------------------------------------------------
161  # Step 2: Install static routes to enforce the desired path.
162  #
163  # Forward path: UT(s) -> SAT1 -> GW -> NCC
164  # Reverse path: NCC -> GW -> SAT1 -> UT(s)
165  # ----------------------------------------------------------------
166 
167  # Each UT: to reach NCC's subnet (10.0.3.0/24), send to SAT1 (10.0.1.2)
168  for ut in uts:
169  ut.cmd('route add -net 10.0.3.0 netmask 255.255.255.0 gw 10.0.1.2')
170 
171  # SAT1: to reach NCC specifically (10.0.3.2), forward to GW (10.0.2.2)
172  sat1.cmd('route add -host 10.0.3.2 gw 10.0.2.2')
173 
174  # NCC: to reply to any UT in subnet 10.0.1.0/24, send back to GW (10.0.3.1)
175  ncc.cmd('route add -net 10.0.1.0 netmask 255.255.255.0 gw 10.0.3.1')
176 
177  # GW: to reply to each UT individually, forward back to SAT1 (10.0.2.1)
178  for i in range(num_uts):
179  ut_ip = f'10.0.1.{10 + i}'
180  gw.cmd(f'route add -host {ut_ip} gw 10.0.2.1')
