147 double
148 RequestManager::DoRbdc(uint8_t rc, FlowType queueType, CapacityRequestCtrlMsg* cr)
149 {
150  double rbdcKbps = 0;
151 
152  double statDuration = Time::Now() - m_llc->GetQueue(queueType)->GetLastStatTime(); // traffic statistic duration
153  double inputRateKbps = m_llc->GetQueue(queueType)->GetInBitsRate(statDuration); // kbps
154  double inputKbits = m_llc->GetQueue(queueType)->GetInBytes()*8; // bits
155  double previousRbdcInKbits = GetPendingRbdcSumKbps(rc)*statDuration;
156  double queueSizeInKbits = m_llc->GetQueue(queueType)->GetQueueSizeInBytes()*8/1000; // kbits
157  double queueOccupancy = std::max(0.0,(queueSizeInKbits - inputKbits - previousRbdcInKbits)/statDuration);
158 
159  double bitsRateKbps = inputRateKbps + queueOccupancy;
160 
161  if(bitsRateKbps <= m_lls->GetCraInKbps(rc))
162  {
163  rbdcKbps = 0;
164  }
165  else
166  {
167  rbdcKbps = bitsRateKbps - m_lls->GetCraInKbps(rc);
168  }
169 
170  if (rbdcKbps + m_lls->GetCraInKbps(rc) > m_lls->GetMaxServiceRate(rc))
171  {
172  rbdcKbps = m_lls->GetMaxServiceRate(rc) - m_lls->GetCraInKbps(rc);
173  }
174 
175  uint16_t quantRbdc = GetQuantizedRbdcValue(rc,rbdcKbps);
176 
177  UpdatePendingRbdcCounters(rc,static_cast<uint32_t>(quantRbdc));
178 
179  cr->CreateMsg(rc,CapacityCategory::DA_RBDC,quantRbdc);
180 
181  return m_lls->GetCraInKbps(rc) + static_cast<double>(quantRbdc);
182 }
