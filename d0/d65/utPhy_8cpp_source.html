117 // Transmits a packet to satellite or NCC with thread-safe sending and bits collection
118 // Uses mutex to prevent concurrent sends. Prepends 4-byte length header to packet.
119 // Ensures all bytes are sent (handles partial sends with retry loop)

122 UtPhy::Transmit(std::vector<uint8_t> buf, bool isTcPkt)
123 {
124  std::lock_guard<std::mutex> lock(m_tx_mutex);
125 
126  std::string rx_name = (isTcPkt && !m_isTcPktViaSat) ? "NCC" : "Sat";
127  int sock = (isTcPkt && !m_isTcPktViaSat) ? m_ncc_sock : m_sock;
128 
129  if (!IsTxPossible()) {
130  m_log->WriteToFile("[UT PHY] Tx not possible, dropping packet.\n");
131  return;
132  }
133 
134  uint32_t dataLen = static_cast<uint32_t>(buf.size());
135  // 1. Convert to network byte order (Big-Endian)
136  // uint32_t netLen = htonl(dataLen);
137  uint32_t netLen = dataLen;
138 
139  // 2. Prepare complete send buffer (4 bytes header + payload)
140  std::vector<uint8_t> sendBuf(sizeof(netLen) + buf.size());
141 
142  // 3. Use memcpy to ensure first 4 bytes are correct
143  std::memcpy(sendBuf.data(), &netLen, sizeof(netLen));
144 
145  // 4. Copy remaining payload
146  if (!buf.empty()) {
147  std::memcpy(sendBuf.data() + sizeof(netLen), buf.data(), buf.size());
148  }
149 
150  // 5. Ensure all bytes are sent
151  size_t totalSize = sendBuf.size();
152  size_t sent = 0;
153  while (sent < totalSize) {
154  ssize_t n = send(sock, sendBuf.data() + sent, totalSize - sent, 0);
155  if (n <= 0) {

157  return;
158  }
159  sent += n;
160  }
161

164 }
