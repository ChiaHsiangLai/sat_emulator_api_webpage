14 // Tracks per-UE turn order for round-robin service. Knows nothing about what a
15 // "turn" produces — callers pull the next UE via Next() and report back what
16 // happened via Report(), both plain values (no callback). One round is
17 // Size() consecutive Next()/Report() pairs; callers drive the round loop
18 // themselves. Not thread-safe; callers serialize access (RawQueue does so
19 // under its own mutex).
20 class RoundRobinScheduler
21 {
22  public:
23  // What happened when a UE's turn was attempted, decided entirely by the caller.
24  enum class Result
25  {
26  Served, // something was served this turn; goes to the back — it just had its turn
27  Skipped, // nothing served this turn (e.g. no room); goes to the front — it's owed a turn
28  Drained, // UE has nothing left; removed from rotation until re-activated
29  };
30 
31  // Adds ueId to the back of the rotation. Callers must only call this on an
32  // empty-to-non-empty transition for that UE — this class does not itself
33  // dedupe against a UE already present in the rotation.
34  void Activate(int ueId) { m_order.push_back(ueId); }
35 
36  // True when no UE is currently in rotation.
37  bool Empty() const { return m_order.empty(); }
38 
39  // Number of UEs currently in rotation; one round is this many Next()/Report() pairs.
40  size_t Size() const { return m_order.size(); }
41 
42  // Pops and returns the UE due for the next turn. Caller must follow up
43  // with exactly one Report() for this ueId before the rotation reflects it again.
44  int Next()
45  {
46  int ueId = m_order.front();
47  m_order.pop_front();
48  return ueId;
49  }
50 
51  // Requeues ueId to the back on Served, to the front on Skipped, drops it
52  // on Drained. The front/back split matters: if Skipped were also sent to
53  // the back, a full round (every current member popped-and-requeued once,
54  // none Drained) restores the exact starting order — so whichever UE
55  // starts at the front stays there and wins every future opportunity
56  // outright, starving the rest, whenever the byte budget only ever fits
57  // one UE's buffer per round. Requeuing Skipped to the front instead means
58  // it leads the very next round/opportunity, so a UE that lost out this
59  // time gets first crack next time instead of being stuck permanently
60  // behind whoever already got served.
61  void Report(int ueId, Result result)
62  {
63  if (result == Result::Served) m_order.push_back(ueId);
64  else if (result == Result::Skipped) m_order.push_front(ueId);
65  }
66 
67  private:
68  std::deque<int> m_order; // rotation order; front is served next
69 };
