1 #!/usr/bin/env python3
2 """Convert iperf2 UDP client or server output to the two-column
3 (unix_timestamp rate_kbps) format expected by time_vs_data_rate_plot.py.
4 
5 Usage:
6  python3 parse_iperf_log.py <iperf_log> <start_time> <output_file> [window_size [apply_ma]]
7 
8  window_size : optional integer, number of seconds for centered moving-average
9  smoothing (default 1 = no smoothing).
10  apply_ma : optional 0/1 or false/true — enable moving-average (default 1).
11  Set to 0 to output raw per-second rates regardless of window_size.
12 
13 Handled line formats (client and server, with or without -f m):
14  Server: [ 1] 22.0000-23.0000 sec 0.191 MBytes 1.61 Mbits/sec 88.873 ms 0/98 (0%)
15  Client: [ 3] 0.0- 1.0 sec 128 KBytes 1.05 Mbits/sec
16  Client: [ 3] 0.0- 1.0 sec 0.125 MBytes 1.05 Mbits/sec
17 
18 Lines may optionally be prefixed with an absolute Unix timestamp (injected by
19 prepend_timestamp.py). When present the prefix is used directly as the output
20 timestamp instead of start_time + relative_t_end, eliminating any error from
21 the iperf_start_time.txt anchor.
22 
23  Prefixed: 1779699891.669040 [ 5] 1.0000-2.0000 sec ...
24 
25 Skipped lines: 'out-of-order' annotations, 'Server Report' headers,
26  'Sent X datagrams' summaries, and full-run summaries
27  (interval duration > 1.5 s treated as the final summary).
28 """
29 import re
30 import sys
31 
32 if len(sys.argv) not in (4, 5, 6):

34  sys.exit(1)

36 iperf_log = sys.argv[1]
37 start_time = float(sys.argv[2])
38 output_file = sys.argv[3]
39 window_size = int(sys.argv[4]) if len(sys.argv) >= 5 else 1
40 apply_ma = sys.argv[5].lower() not in ('0', 'false', 'no') if len(sys.argv) == 6 else True

42 # Matches any per-1-second interval line from both client and server output.
43 # Handles Kbits/sec, Mbits/sec, and Gbits/sec.
44 pattern = re.compile(
45  r'\[\s*\d+\]\s+([\d.]+)\s*-\s*([\d.]+)\s+sec\s+[\d.]+\s+\S+\s+([\d.]+)\s+(K|M|G)bits/sec'
46 )
47 
48 # Detects an absolute Unix timestamp prefix injected by prepend_timestamp.py.
49 # Matches e.g. "1779699891.669040 " at the start of a line.
50 abs_ts_re = re.compile(r'^(\d{10,}(?:\.\d+)?)\s+')
51 
52 records = [] # (timestamp, rate_kbps, duration_s)
53 time_offset = 0.0 # accumulated offset across chained iperf segments (VBR mode)
54 last_t_end = 0.0
55 try:
56  with open(iperf_log) as f:
57  for line in f:
58  if 'out-of-order' in line:
59  continue
60  if 'Server Report' in line or 'Sent' in line:
61  continue
62  m = pattern.search(line)
63  if m:
64  t_start = float(m.group(1))
65  t_end = float(m.group(2))
66  duration = t_end - t_start
67  # Skip the final full-run summary: starts from 0 and spans many seconds.
68  # (Intermediate long intervals caused by TBTP bursty delivery are kept.)
69  if t_start < 0.05 and duration > 1.5:
70  continue
71  # Skip spurious sub-second sessions caused by TBTP burst backlog
72  # forwarded after the main test ends (iperf server WARNING sessions).
73  if t_start < 0.5 and duration < 0.5:
74  continue
75  # Detect segment restart: t_start jumped back (chained iperf VBR segments).
76  # A small tolerance avoids false positives from formatting/rounding noise,
77  # while still catching repeated 0.0-1.0 one-second iperf segments.
78  if t_start < last_t_end - 0.05:
79  time_offset += last_t_end
80  last_t_end = t_end
81  rate = float(m.group(3))
82  unit = m.group(4) # 'K', 'M', or 'G'
83  rate_kbps = rate if unit == 'K' else rate * 1000.0 if unit == 'M' else rate * 1_000_000.0
84  abs_m = abs_ts_re.match(line)
85  timestamp = (float(abs_m.group(1)) if abs_m
86  else start_time + time_offset + t_end)
87  records.append((timestamp, rate_kbps, duration))
88 except FileNotFoundError:
89  print(f"Error: file not found: {iperf_log}")
90  sys.exit(1)
91 
92 if not records:
93  print("Warning: no iperf interval lines found in log.")
94  sys.exit(1)

96 # Apply trailing moving-average smoothing (duration-weighted) if enabled.
97 # Window covers [i-window_size+1 .. i], so the smoothed value is anchored to
98 # the most-recent sample's timestamp with no look-ahead.
99 # Using duration weights correctly handles variable-length intervals from TBTP-bursty
100 # satellite delivery, where iperf server may report long low-rate gaps between bursts.
101 if apply_ma and window_size > 1:
102  smoothed = []
103  for i in range(len(records)):
104  lo = max(0, i - window_size + 1)
105  hi = i + 1
106  total_kbits = sum(r * d for _, r, d in records[lo:hi])
107  total_dur = sum(d for _, _, d in records[lo:hi])
108  avg = total_kbits / total_dur if total_dur > 0 else 0.0
109  smoothed.append((records[i][0], avg))
110  records = smoothed
111 else:
112  records = [(t, r) for t, r, _ in records]
113 
114 with open(output_file, 'w') as f:
115  for t, r in records:
116  f.write(f"{t:.5f} {r:.5f}\n")
