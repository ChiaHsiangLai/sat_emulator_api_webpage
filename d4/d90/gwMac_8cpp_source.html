440 void
441 GwMac::CheckLinkStatusAndCnt()
442 {
443  bool last_link_up = false;
444  std::string last_processed_addr = "";
445  int down_count = 0;
446  const int DOWN_THRESHOLD = 5;
447 
448  while (1)
449  {
450  // if there is an ongoing connection process (either handover or reconnect), skip link status check to avoid false detection during handover/reconnect process.
451  if (m_isConnecting)
452  {
453  Time::Sleep(1.0);
454  continue;
455  }
456 
457  std::string current_addr = m_nowSatAddr;
458 
459  if (current_addr != last_processed_addr)
460  {
461  std::cout << "[GW MAC] Target satellite changed to " << current_addr << "\n";
462  last_link_up = false;
463  down_count = 0;
464  last_processed_addr = current_addr;
465  }
466 
467  bool link_signal_detected = m_phy->IsLinkUp(current_addr);
468 
469  if (link_signal_detected)
470  {
471  down_count = 0;
472  if (!last_link_up)
473  {
474  std::cout << "[GW MAC] Sat " << current_addr << " link UP !!\n";
475  m_phy->SetTxSatState(true);
476  last_link_up = true;
477  }
478 
479  // process pending handover messages if the link is up
480  if(!m_hoMsgContainer.empty() && m_HoEnabled)
481  {
482  std::string targetSatIp = m_hoMsgContainer.back()->GetData()->handoverMessage;
483  std::cout << "[GW MAC] Command: Handover to " << targetSatIp << "\n";
484  m_hoMsgContainer.clear();
485 
486  SendHandoverNotificationMsg(m_phy->GetSatSock());
487  Handover(targetSatIp);
488  continue;
489  }
490  }
491  else
492  {
493  // process connection recovery if the link is down, start counting.
494  // If the link recovers before reaching the threshold, reset the count. If the link is still down after reaching the threshold, trigger handover process.
495  if (last_link_up)
496  {
497  down_count++;
498  if (down_count >= DOWN_THRESHOLD && m_HoEnabled)
499  {
500  std::cout << "[GW MAC] Link LOST. Triggering recovery...\n";
501  m_phy->SetTxSatState(false);
502  last_link_up = false;
503 
504  SendHandoverNotificationMsg(m_phy->GetSatSock());
505  StartHandoverProcess(current_addr);
506 
507  down_count = 0;
508  }
509  }
510  }
511  Time::Sleep(0.5);
512  }
513 }
