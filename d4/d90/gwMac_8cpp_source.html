439 void
440 GwMac::CheckLinkStatusAndCnt()
441 {
442  bool last_link_up = false;
443  std::string last_processed_addr = "";
444  int down_count = 0;
445  const int DOWN_THRESHOLD = 5;
446 
447  while (1)
448  {
449  // if there is an ongoing connection process (either handover or reconnect), skip link status check to avoid false detection during handover/reconnect process.
450  if (m_isConnecting)
451  {
452  Time::Sleep(1.0);
453  continue;
454  }
455 
456  std::string current_addr = m_nowSatAddr;
457 
458  if (current_addr != last_processed_addr)
459  {
460  std::cout << "[GW MAC] Target satellite changed to " << current_addr << "\n";
461  last_link_up = false;
462  down_count = 0;
463  last_processed_addr = current_addr;
464  }
465 
466  bool link_signal_detected = m_phy->IsLinkUp(current_addr);
467 
468  if (link_signal_detected)
469  {
470  down_count = 0;
471  if (!last_link_up)
472  {
473  std::cout << "[GW MAC] Sat " << current_addr << " link UP !!\n";
474  m_phy->SetTxSatState(true);
475  last_link_up = true;
476  }
477 
478  // process pending handover messages if the link is up
479  if(!m_hoMsgContainer.empty() && m_HoEnabled)
480  {
481  std::string targetSatIp = m_hoMsgContainer.back()->GetData()->handoverMessage;
482  std::cout << "[GW MAC] Command: Handover to " << targetSatIp << "\n";
483  m_hoMsgContainer.clear();
484 
485  SendHandoverNotificationMsg(m_phy->GetSatSock());
486  Handover(targetSatIp);
487  continue;
488  }
489  }
490  else
491  {
492  // process connection recovery if the link is down, start counting.
493  // If the link recovers before reaching the threshold, reset the count. If the link is still down after reaching the threshold, trigger handover process.
494  if (last_link_up)
495  {
496  down_count++;
497  if (down_count >= DOWN_THRESHOLD && m_HoEnabled)
498  {
499  std::cout << "[GW MAC] Link LOST. Triggering recovery...\n";
500  m_phy->SetTxSatState(false);
501  last_link_up = false;
502 
503  SendHandoverNotificationMsg(m_phy->GetSatSock());
504  StartHandoverProcess(current_addr);
505 
506  down_count = 0;
507  }
508  }
509  }
510  Time::Sleep(0.5);
511  }
