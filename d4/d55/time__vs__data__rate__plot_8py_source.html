89 def parse_iperf_cwnd(log_file):
90  """Return list of (unix_timestamp, cwnd_kbytes, retr) from iperf3 log."""
91  records = []
92  data_re = re.compile(r'^\d+\.\d+\s+\[\s*\d+\]')
93  with open(log_file, 'r') as f:
94  for line in f:
95  line = line.strip()
96  if not data_re.match(line):
97  continue
98  parts = line.split()
99  # Need at least: ts [N] interval sec xfer unit rate runit retr cwnd cunit
100  if len(parts) < 11:
101  continue
102  try:
103  ts = float(parts[0])
104  cwnd_unit = parts[-1].lower()
105  cwnd_val = float(parts[-2])
106  retr = int(parts[-3])
107  cwnd_kb = cwnd_val * _CWND_UNIT_TO_KB.get(cwnd_unit, 1.0)
108  records.append((ts, cwnd_kb, retr))
109  except (ValueError, IndexError):
110  continue
111  return records
