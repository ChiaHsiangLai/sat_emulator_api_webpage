1 #!/usr/bin/env python3
2 """
3 hls_live_push_client.py — the uplink-push counterpart to hls_live_client.py:
4 instead of a viewer pulling segments from a remote server, this simulates a
5 live broadcaster actively pushing its own growing HLS stream out to an
6 ingest server (python_tools/hls_live_ingest_server.py) as segments are
7 produced — the way a real streaming app (going live from a phone) initiates
8 the upload itself, rather than passively waiting for someone to connect in.
9 
10 Reads the local HLS playlist directly off disk instead of fetching it over
11 HTTP — this runs on the same node as the ffmpeg -re live encoder (see
12 launch_live_video_encoder), so there's no need to round-trip a GET just to
13 discover which segments exist. As new segments appear, each is POSTed to
14 <ingest_base_url>/segment/<name>; after any successful push, a *synced*
15 playlist — built from only the segment names that have actually finished
16 uploading so far (see build_synced_playlist) — is POSTed to
17 <ingest_base_url>/playlist. This is deliberately NOT the same text as the
18 local playlist file: the local file can already list segments the encoder
19 has produced but this client hasn't gotten to (or failed to) push yet
20 — e.g. when push falls behind under a congested uplink and several
21 segments pile up between polls, or when <duration_s> is hit mid-batch —
22 and forwarding it verbatim would make the ingest server advertise segments
23 that 404 when a viewer actually requests them. #EXT-X-ENDLIST is only
24 appended once the local playlist has it AND every segment the local
25 playlist currently lists has been confirmed pushed.
26 
27 PUSH LATENCY: each pushed segment's log line reports how far behind the
28 live edge the push itself finished — i.e. how much the uplink adds on top
29 of real-time production. This is (time push finished) - (segment file's
30 own local mtime, which is "when the encoder finished producing this
31 segment" since ffmpeg -re paces writes at real playback speed).
32 
33 Runs until the encoder's playlist reaches #EXT-X-ENDLIST (and that final
34 playlist state has been pushed) or <duration_s> elapses, whichever comes
35 first. On exit, writes a small key=value summary to <summary_path>.
36 
37 Usage:
38  hls_live_push_client.py <local_playlist_path> <ingest_base_url> <duration_s> <summary_path>
39  e.g. hls_live_push_client.py data/live_video_ue101/live.m3u8 \\
40  http://10.0.255.254:5101 300 log_file/UE_10.0.200.101/video_live_push_duration.txt
41 """

47 CONNECT_TIMEOUT_S = 60 # see hls_live_client.py — measured ut_queueing_delay
48 # averages ~15.6s and peaks ~70s under congestion, so a short connect-timeout
49 # just guarantees most connection attempts get killed before ever getting a
50 # fair chance, no matter how many times they're retried.

52 STALL_WINDOW_S = 30 # only after this many seconds sustained below that speed
53 POLL_INTERVAL_S = 1.0 # how often to re-check the local playlist for new segments

78 def build_synced_playlist(playlist_text, pushed_set, append_endlist):
79  """Return a version of playlist_text containing only the header
80  directives plus #EXTINF/segment-name pairs for segments actually in
81  pushed_set, in their original order — so the ingest server never
82  advertises a segment that hasn't actually finished uploading (see the
83  module docstring for why the raw local playlist can't be forwarded
84  as-is). #EXT-X-ENDLIST is appended only if append_endlist is True."""
85  out = []
86  pending_extinf = None
87  for line in playlist_text.splitlines():
88  stripped = line.strip()
89  if not stripped:
90  continue
91  if stripped.startswith("#EXTINF:"):
92  pending_extinf = stripped
93  elif stripped == ENDLIST_MARKER:
94  continue # only appended at the end, and only if append_endlist
95  elif not stripped.startswith("#"):
96  if stripped in pushed_set:
97  if pending_extinf is not None:
98  out.append(pending_extinf)
99  out.append(stripped)
100  pending_extinf = None
101  else:
102  out.append(stripped) # header directives: #EXTM3U, #EXT-X-VERSION, etc.
103  if append_endlist:
104  out.append(ENDLIST_MARKER)
105  return "\n".join(out) + "\n"

108 def push_segment(ingest_base_url, seg_path, seg_name, timeout_s):
109  """POST seg_path's contents to the ingest server. Returns (elapsed_s, success).
110 
111  --fail makes curl treat a non-2xx HTTP response as a failure. A real
112  stall (sustained near-zero throughput) or a hard hang past timeout_s
113  also counts as a failure — same reasoning as hls_live_client.py's
114  fetch_segment."""
115  t0 = time.time()
116  try:
117  result = subprocess.run(
118  ["curl", "-s", "-f", "--connect-timeout", str(CONNECT_TIMEOUT_S),
119  "--speed-limit", str(STALL_SPEED_BYTES_PER_SEC), "--speed-time", str(STALL_WINDOW_S),
120  "-X", "POST", "--data-binary", f"@{seg_path}",
121  f"{ingest_base_url}/segment/{seg_name}", "-o", "/dev/null"],
122  timeout=timeout_s,
123  )
124  except subprocess.TimeoutExpired:
125  return time.time() - t0, False
126  return time.time() - t0, (result.returncode == 0)

129 def push_playlist(ingest_base_url, playlist_text, timeout_s):
130  """POST the current full playlist text to the ingest server. Returns success bool."""
131  try:
132  result = subprocess.run(
133  ["curl", "-s", "-f", "--connect-timeout", str(CONNECT_TIMEOUT_S),
134  "-X", "POST", "--data-binary", "@-", f"{ingest_base_url}/playlist", "-o", "/dev/null"],
135  input=playlist_text, capture_output=True, text=True, timeout=timeout_s,
136  )
137  except subprocess.TimeoutExpired:
138  return False
139  return result.returncode == 0

142 def main():
143  if len(sys.argv) != 5:
144  print("Usage: hls_live_push_client.py <local_playlist_path> <ingest_base_url> "
145  "<duration_s> <summary_path>", file=sys.stderr)
146  sys.exit(1)
147 
148  playlist_path = sys.argv[1]
149  ingest_base_url = sys.argv[2].rstrip("/")
150  run_duration = float(sys.argv[3])
151  summary_path = sys.argv[4]
152  base_dir = os.path.dirname(playlist_path)
153 
154  start = time.time()
155  seen = set()
156  pushed_set = set() # segments CONFIRMED successfully pushed — the only ones ever advertised
157  pushed = 0
158  failed = 0
159  saw_playlist = False
160  playlist_synced = False # whether the ingest server's playlist already matches pushed_set's final state
161  latencies = []
162 
163  while time.time() - start < run_duration:
164  remaining = run_duration - (time.time() - start)
165  if not os.path.exists(playlist_path):
166  print(f"[hls_live_push_client] local playlist not yet at {playlist_path}, retrying in 1s", flush=True)
167  time.sleep(min(1.0, max(remaining, 0)))
168  continue
169  with open(playlist_path) as f:
170  playlist_text = f.read()
171  if not playlist_text or "#EXTM3U" not in playlist_text:
172  time.sleep(min(1.0, max(remaining, 0)))
173  continue
174  saw_playlist = True
175 
176  stream_ended = ENDLIST_MARKER in playlist_text
177  seg_names = parse_segment_names(playlist_text)
178  new_names = [n for n in seg_names if n not in seen]
179 
180  for seg_name in new_names:
181  if time.time() - start >= run_duration:
182  break
183  seg_path = os.path.join(base_dir, seg_name)
184  if not os.path.exists(seg_path):
185  continue # ffmpeg listed it in the playlist just before finishing the write — catch it next poll
186  seen.add(seg_name)
187  seg_timeout = CONNECT_TIMEOUT_S + max(run_duration - (time.time() - start), 0)
188  produced_at = os.path.getmtime(seg_path)
189  push_time, success = push_segment(ingest_base_url, seg_path, seg_name, seg_timeout)
190  if success:
191  pushed += 1
192  pushed_set.add(seg_name)
193  lag_s = time.time() - produced_at
194  latencies.append(lag_s)
195  print(f"[hls_live_push_client] pushed {seg_name} (took {push_time:.2f}s, "
196  f"push latency: {lag_s:.1f}s behind)", flush=True)
197 
198  # Advertise this segment to viewers the instant it lands,
199  # rather than waiting for the rest of new_names to finish —
200  # under a congested uplink the encoder can produce several
201  # more segments while a single push_segment call is still in
202  # flight (each one here can take many seconds), so by the
203  # time this for-loop would otherwise finish, new_names may
204  # have grown to cover many segments at once. Syncing here
205  # instead of after the loop is what keeps the viewer
206  # advancing one segment at a time as each genuinely arrives,
207  # instead of stalling through a whole pile-up and then
208  # jumping forward by several segments in one go.
209  fully_delivered_now = stream_ended and all(n in pushed_set for n in seg_names)
210  synced_text = build_synced_playlist(playlist_text, pushed_set, append_endlist=fully_delivered_now)
211  if push_playlist(ingest_base_url, synced_text, CONNECT_TIMEOUT_S + 10):
212  playlist_synced = fully_delivered_now
213  else:
214  print("[hls_live_push_client] failed to push updated playlist to ingest server", flush=True)
215  else:
216  failed += 1
217  print(f"[hls_live_push_client] failed to push {seg_name} "
218  f"(curl error, connect-timeout, or stall)", flush=True)
219 
220  # Fully delivered only once the encoder has said it's done AND every
221  # segment it has ever listed has actually been confirmed pushed — a
222  # segment that failed above, or one skipped this round because
223  # run_duration ran out mid-batch, must NOT count as delivered. This
224  # only needs to actively push anything when new_names was empty (no
225  # segment above already just synced the final ENDLIST state).
226  fully_delivered = stream_ended and all(n in pushed_set for n in seg_names)
227  if not new_names and fully_delivered and not playlist_synced:
228  synced_text = build_synced_playlist(playlist_text, pushed_set, append_endlist=True)
229  if push_playlist(ingest_base_url, synced_text, CONNECT_TIMEOUT_S + 10):
230  playlist_synced = True
231  else:
232  print("[hls_live_push_client] failed to push updated playlist to ingest server", flush=True)
233 
234  if not new_names:
235  if fully_delivered and playlist_synced:
236  print("[hls_live_push_client] live stream ended and every segment confirmed pushed, done", flush=True)
237  break
238  time.sleep(min(POLL_INTERVAL_S, max(remaining, 0)))
239 
240  elapsed = time.time() - start
241  if not saw_playlist:
242  print(f"[hls_live_push_client] local playlist never appeared at {playlist_path}, giving up", flush=True)
243  write_summary(summary_path, "no_playlist", elapsed, 0, 0)
244  return
245 
246  latency_last = latencies[-1] if latencies else None
247  latency_avg = sum(latencies) / len(latencies) if latencies else None
248  latency_max = max(latencies) if latencies else None
249 
250  if elapsed >= run_duration:
251  status = "timed_out"
252  print(f"[hls_live_push_client] run_duration ({run_duration:.0f}s) exceeded before the live stream "
253  f"finished pushing ({pushed} pushed, {failed} failed), stopping early", flush=True)
254  else:
255  status = "completed" if failed == 0 else "completed_with_errors"
256  print(f"[hls_live_push_client] finished pushing live stream "
257  f"({pushed} pushed, {failed} failed) in {elapsed:.2f}s", flush=True)
258  if latencies:
259  print(f"[hls_live_push_client] push latency — last: {latency_last:.1f}s, "
260  f"avg: {latency_avg:.1f}s, max: {latency_max:.1f}s", flush=True)
261 
262  write_summary(summary_path, status, elapsed, pushed, segments_failed=failed,
263  latency_last_s=latency_last, latency_avg_s=latency_avg, latency_max_s=latency_max)
