54 {

56 
57  // A single TCP message from a UT may contain multiple concatenated serialized
58  // packets (e.g. two TC segments that were batched into one TBTP slot).
59  // Loop until all bytes in buf are consumed so no packet is silently dropped.
60  size_t offset = 0;
61  while (offset < buf.size())
62  {
63  std::vector<uint8_t> slice(buf.begin() + offset, buf.end());
64  Packet* pkt = new Packet();
65  if (!pkt->Deserialize(slice))
66  {
67  delete pkt;
68  break;
69  }
70  uint32_t consumed = pkt->GetPktSizeBytes();
71  if (consumed == 0) { delete pkt; break; }
72  offset += consumed;
73 
74  // *** [QUEUE SPLIT] ***
75  // TBTPs go into the dedicated m_tbtpRxQueue; all other signaling (including DISCONNECT) goes into m_rxQueue.
76  // Each queue is processed by an independent thread, ensuring DISCONNECT cannot jump ahead of pending TBTPs.
77  CtrlMsgTag* ctrlTag = pkt->PeekTag<CtrlMsgTag>();
78  if (ctrlTag && ctrlTag->GetMsgType() == CtrlMsgType::TBTP_CTRL_MSG && m_tbtpProcRunning)
79  {
80  std::lock_guard<std::mutex> lock(m_tbtpRxQueueMtx);
81  m_tbtpRxQueue.push(pkt);
82  m_tbtpRxCv.notify_one();
83  }
84  else
85  {
86  std::lock_guard<std::mutex> lock(m_rxQueueMtx);
87  m_rxQueue.push(pkt);
88  m_pktRxCv.notify_one();
89  }
90 
91  if(pkt->PeekTag<TrcTag>()) continue;
92 
93  if (ctrlTag && ctrlTag->GetMsgType() == CtrlMsgType::TBTP_CTRL_MSG)
94  {
95  uint32_t pkt_size_bytes = pkt->GetPktSizeBytes();
96  double tbtp_prop_time = tbtp_rev_time - pkt->PeekTag<SendTimeTag>()->GetTime();
97  std::ostringstream oss;

100  }
101 
102  // statistic packet bits receive, write to file

521 // ISL forwarding task: wait until all handing-over UTs complete, then forward buffered TBTPs to ISL in order.
522 void
523 SatMac::FlushTbtpBufferToIslTask()
524 {
525  while (m_tbtpFlushRunning)
526  {
527  std::unique_lock<std::mutex> lock(m_tbtpBufferMtx);
528 
529  // Wait until all expected UTs have completed handover (m_expectedHandoverCount reset to 0) and the buffer is non-empty
530  m_tbtpBufferCv.wait(lock, [&] {
531  return (m_expectedHandoverCount == 0 && !m_tbtpBuffer.empty()) || !m_tbtpFlushRunning;
532  });
533 
534  if (!m_tbtpFlushRunning)
535  break;
536 
537  // Drain all buffered TBTPs into a local queue, then release the lock
538  std::queue<Packet*> localQueue;
539  std::swap(localQueue, m_tbtpBuffer);
540  lock.unlock();
541 
542  std::ostringstream oss;
543  oss << "[SAT MAC] Flushing " << localQueue.size()
544  << " buffered TBTPs to ISL.\n";
545  m_log->WriteToFile(oss.str());
546 
547  while (!localQueue.empty())
548  {
549  Packet* pkt = localQueue.front();
550  localQueue.pop();
551 
552  MacAddressTag* macAddrTag = pkt->PeekTag<MacAddressTag>();
553  macAddrTag->SetSourceAddrVal("1.1.1.1");
554 
555  std::vector<uint8_t> buf;
556  pkt->Serialize(buf);
557 
558  m_islPhy->Broadcast(buf);
559  }
560 
561  oss.str("");
562  oss << "[SAT MAC] All buffered TBTPs have been forwarded to ISL.\n";
563  m_log->WriteToFile(oss.str());

647 {

650  Packet* pkt = nullptr;
651  int sock = -1;
652 
653  {
654  std::unique_lock<std::mutex> lock(m_rxQueueMtx);
655 
656  m_pktRxCv.wait(lock, [&] {
657  return !m_rxQueue.empty() || !m_pktProcRunning;
658  });
659 
660  if (!m_pktProcRunning && m_rxQueue.empty())
661  return;
662 
663  pkt = m_rxQueue.front();
664  m_rxQueue.pop();
665  sock = 1;
666  }
667 
668  if (!pkt)
669  continue;
670 
671  // peek address
672  MacAddressTag* macAddrTag = pkt->PeekTag<MacAddressTag>();
673  Address destAddr(macAddrTag->GetDestinationAddrVal(),macAddrTag->GetAddrType(),macAddrTag->GetDstAddrLen());
674  Address srcAddr(macAddrTag->GetSourceAddrVal(),macAddrTag->GetAddrType(),macAddrTag->GetSrcAddrLen());
675 
676  std::ostringstream oss;
677  oss << "[SAT MAC] Receive packet from address : " << srcAddr.GetAddressVal() << "\n";
678  m_log->WriteToFile(oss.str());
679 
680  // receive function for different packet
681  if (pkt->PeekTag<CtrlMsgTag>())
682  {
683  ReceiveSignalingPacket(sock,pkt);
684  }
685  else if (pkt->PeekTag<TrcTag>())
686  {
687  m_log->ClearOss(oss);
688  oss << "[SAT MAC] Received TC packet message from address : " << macAddrTag->GetSourceAddrVal() << "\n";
689  m_log->WriteToFile(oss.str());
690 
691  GroundStationTag* gwAddrTag = pkt->PeekTag<GroundStationTag>();
692  E2EAddressTag* e2eAddrTag = pkt->PeekTag<E2EAddressTag>();
693 
694  std::string gwAddr = gwAddrTag->GetSourceAddrVal();
695 
696  if (!HasGW(gwAddr))

699  int gwId = GetDevId(gwAddr);

701  m_log->WriteToFile(oss.str());
702 
703  UnicastToSat(pkt);
704  }
